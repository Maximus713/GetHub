<?php

final class NPK_Concentration_Calculation_Engine {
    
    /**
     * Основной расчётный метод (реализация)
     */
    public static function calculate(int $fertilizer_id, float $concentration, float $water_volume, string $ground_type): array {
        // 1. Проверка NPK_Database
        if (!class_exists('NPK_Database')) {
            return self::get_error_result('База данных не загружена');
        }
        
        // 2. Получение удобрения
        $fertilizer = self::get_fertilizer_by_id($fertilizer_id);
        if (!$fertilizer) {
            return self::get_error_result("Удобрение ID {$fertilizer_id} не найдено");
        }
        
        // 3. Коррекция ввода
        $corrected_concentration = self::correct_concentration($concentration);
        $corrected_water_volume = self::correct_water_volume($water_volume);
        
        // 4. Расчёт массы
        $fertilizer_mass = self::calculate_mass(
            $corrected_concentration,
            $corrected_water_volume,
            $fertilizer['solubility_total']
        );
        
        // 5. Определение состояния
        $state = self::determine_state($corrected_concentration, $ground_type);
        
        // 6. Возврат результата
        return [
            'fertilizer_mass' => $fertilizer_mass,
            'concentration' => $corrected_concentration,
            'water_volume' => $corrected_water_volume,
            'state' => $state,
            'warnings' => self::get_warnings($corrected_concentration, $fertilizer, $state)
        ];
    }
    
    /**
     * Определение состояния (публичный метод)
     */
    public static function get_concentration_state(float $concentration, string $ground_type): array {
        $corrected = self::correct_concentration($concentration);
        return self::determine_state($corrected, $ground_type);
    }
    
    /**
     * Валидация растворимости
     */
    protected static function validate_solubility_total($value): float {
        if ($value === null || $value === 0) return 100.0;
        if ($value < 1) return 1.0;
        if ($value > 100) return 100.0;
        return (float)$value;
    }
