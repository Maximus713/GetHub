```python
def базовый_аудит(код, тз):
    # 1. Механическая проверка синтаксиса
    if not компилируется(код):
        исправить_синтаксис(код)  # только очевидные исправления
    
    # 2. Поиск по чётким шаблонам
    проблемы = []
    for шаблон in КРИТИЧЕСКИЕ_ШАБЛОНЫ:
        if найден_шаблон(код, шаблон):
            проблемы.append(создать_исправление(шаблон))
    
    # 3. Проверка соответствия ТЗ (только по именам)
    for функция in тз.обязательные_функции:
        if not функция_существует(код, функция):
            проблемы.append(f"Добавить функцию: {функция}")
    
    # 4. Безопасные исправления
    исправленный_код = код
    for проблема in проблемы:
        патч = создать_минимальный_патч(проблема)
        if безопасен(патч):  # проверка: ≤3 строк, не меняет логику
            исправленный_код = применить(исправленный_код, патч)
    
    return исправленный_код, проблемы
```

### **АЛГОРИТМ РАБОТЫ (адаптивный):**
```python
def адаптивный_аудит(код, тз, конфиг):
    # 1. Расширенный анализ на основе фокуса
    if "валидация_данных" in конфиг.фокус_поиска:
        проблемы += найти_проблемы_валидации(код)
    
    if "отправка_почты" in конфиг.фокус_поиска:
        проблемы += найти_проблемы_почты(код)
    
    # 2. Глубокий анализ с риском
    for проблема in проблемы:
        варианты_исправлений = найти_все_варианты(проблема)
        
        # Перебор ВСЕХ вариантов (как вы хотели)
        for вариант in варианты_исправлений:
            тестируемый_код = применить(код, вариант)
            
            # Мысленный тест на основе описания ошибки
            if проходит_мысленный_тест(тестируемый_код, описание_ошибки):
                if безопасно(вариант, конфиг):
                    добавить_в_план(вариант)
                    break  # нашли рабочий вариант
    
    # 3. Итеративное применение с откатом
    текущий_код = код
    история = []
    
    for исправление in план_исправлений:
        # Сохраняем состояние
        история.append(текущий_код)
        
        # Пробуем применить
        тестовый_код = применить(текущий_код, исправление)
        
        # Расширенная проверка
        if не_ломает_другое(тестовый_код, тз) and \
           решает_проблему(тестовый_код, описание_ошибки):
            текущий_код = тестовый_код
        else:
            # Откат + отметка проблемы
            текущий_код = история.pop()
            пометить_как_рисковое(исправление)
    
    return текущий_код, отчет_об_изменениях
```

### **СИСТЕМА ОТКАТА В РЕАЛЬНОМ ВРЕМЕНИ:**
