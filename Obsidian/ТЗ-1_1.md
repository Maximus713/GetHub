```php
/**

    exit;

    define('NPK_CALCULATORS_PATH', plugin_dir_path(__FILE__));
}

/**
 * Инициализация плагина
 */
function npk_calculators_init() {
    // 1. Сначала загружаем базу данных (обязательно первой!)
    require_once NPK_CALCULATORS_PATH . 'shared/config/npk-database.php';

    if (!class_exists('NPK_Database')) {
        error_log('NPK Calculators: Не удалось загрузить базу данных');
        return;
    }
    
    // 2. Потом загружаем модуль концентрации
    require_once NPK_CALCULATORS_PATH . 'module-concentration/concentration-module.php';
    
    // 3. Инициализируем модуль (если класс существует)
    if (class_exists('NPK_Concentration_Module')) {
        NPK_Concentration_Module::get_instance();
    } else {
        error_log('NPK Calculators: Не удалось загрузить модуль концентрации');
    }
}

// Ранняя инициализация (приоритет 5 на plugins_loaded)
add_action('plugins_loaded', 'npk_calculators_init', 5);
```

**ФАЙЛ: npk-calculators/module-concentration/concentration-module.php**
РОЛЬ: Главный класс модуля концентрации
ЗАДАЧА: Синглтон-класс, регистрирует скрипты, подключает зависимости, инициализирует компоненты
КОД (ШАБЛОН):
```php
<?php
/**
 * Модуль концентрации рабочего раствора
 */

if (!defined('ABSPATH')) {
    exit;
}

final class NPK_Concentration_Module {
    
    /**
     * Экземпляр класса (синглтон)
     */
    private static $instance = null;
    
    /**
     * Получить экземпляр класса
     */
    public static function get_instance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    /**
     * Конструктор (приватный)
     */
    private function __construct() {
        // 1. Сначала регистрируем скрипты и стили (ранний приоритет!)
        add_action('wp_enqueue_scripts', [$this, 'register_assets'], 1);
        
        // 2. Потом загружаем зависимости
        $this->load_dependencies();
        
        // 3. Инициализируем компоненты
        add_action('init', [$this, 'init']);
    }
    
    /**
     * Загрузка зависимостей
     */
    private function load_dependencies() {
        // Загружаем все классы модуля
        $includes_path = plugin_dir_path(__FILE__) . 'includes/';
        
        require_once $includes_path . 'class-calculation-engine.php';
        require_once $includes_path . 'class-data-adapter.php';
        require_once $includes_path . 'class-ajax-handler.php';
        require_once $includes_path . 'class-shortcode.php';
    }
    
    /**
     * Регистрация скриптов и стилей
     */
    public function register_assets() {
        // Регистрация JavaScript
        wp_register_script(
            'npk-concentration-calculator',
            plugin_dir_url(__FILE__) . 'assets/js/calculator.js',
            [],
            '1.0.0',
            true
        );
        
        // Регистрация CSS
        wp_register_style(
            'npk-concentration-calculator',
            plugin_dir_url(__FILE__) . 'assets/css/calculator.css',
            [],
            '1.0.0'
        );
    }
    
    /**
     * Инициализация компонентов
     */
    public function init() {
        // Проверяем, что все классы загружены
        if (!class_exists('Shortcode')) {
            error_log('NPK Concentration Module: Класс Shortcode не найден');
            return;
        }
        
        // Регистрируем шорткод
        add_shortcode('npk_concentration_calculator', ['Shortcode', 'render']);
        
        // Инициализируем AJAX обработчик
        if (class_exists('AJAX_Handler')) {
            AJAX_Handler::init();
        }
    }
    
    /**
     * Запрет клонирования
     */
    private function __clone() {}
    
    /**
     * Запрет десериализации
     */
    public function __wakeup() {
        throw new Exception("Cannot unserialize singleton");
    }
}
```

```javascript
                const nameElement = stateElement.querySelector('[data-npk-state-name]');
                
                if (emojiElement) emojiElement.textContent = results.state.emoji;
                if (nameElement) nameElement.textContent = results.state.name;
                
                // Устанавливаем цвет состояния
                stateElement.style.backgroundColor = results.state.color;

                sliderFill.style.backgroundColor = results.state.color;
```
