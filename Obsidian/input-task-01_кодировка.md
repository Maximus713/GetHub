<?php
class NPK_Fertilizer_Selection_Engine {
    
    public static function find_matching_fertilizers(string $crop_id, string $phase_id): array {
        $requirements = NPK_Database::get_crop_phase_requirements($crop_id, $phase_id);
        if (!$requirements) {
            return [];
        }
        
        $target_npk = $requirements['npk'];
        $priority = $requirements['priority'] ?? [];
        $fertilizers = NPK_Database::get_fertilizers();
        $threshold = NPK_Database::get_constants()['thresholds']['min_match_percentage'];
        
        $results = [];
        
        foreach ($fertilizers as $fertilizer) {
            $fert_relative = array_map(fn($p) => $p / 100, $fertilizer['npk']);
            
            $norm_target = NPK_Database::normalize_npk($target_npk);
            $norm_fert = NPK_Database::normalize_npk($fert_relative);
            
            $overlap = 0;
            for ($i = 0; $i < 3; $i++) {
                $overlap += min($norm_target[$i], $norm_fert[$i]);
            }
            
            $percentage = round($overlap * 100, 1);
            
            $tips = [];
            if (!empty($priority) && !empty($fertilizer['additives'])) {
                $intersection = array_intersect($priority, $fertilizer['additives']);
                foreach ($intersection as $element) {
                    $tips[] = "Содержит {$element}, важный для {$requirements['phase_name']}";
                }
            }
            
            $results[] = [
                'id' => $fertilizer['id'],
                'name' => $fertilizer['name'],
                'npk' => $fertilizer['npk'],
                'match_percentage' => $percentage,
                'tips' => $tips,
                'fallback' => false
            ];
        }
        
        $passing = array_filter($results, fn($item) => $item['match_percentage'] >= $threshold);
        
        if (count($passing) === 0) {
            usort($results, fn($a, $b) => 
                $b['match_percentage'] <=> $a['match_percentage'] ?: strcmp($a['name'], $b['name'])
            );
            
            $fallback = array_slice($results, 0, 3);
            foreach ($fallback as &$item) {
                $item['fallback'] = true;
            }
            return $fallback;
        }
        
        usort($passing, fn($a, $b) => 
            $b['match_percentage'] <=> $a['match_percentage'] ?: strcmp($a['name'], $b['name'])
        );
        
        return array_values($passing);
    }
}
