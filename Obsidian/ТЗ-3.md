**РОЛЬ:** Адаптер данных для UI. Подготавливает данные из NPK_Database для передачи в JavaScript через `wp_localize_script()`. НЕ участвует в расчётах.

**ЗАДАЧА:** Реализовать все методы согласно ТЗ п.7.3.1 с учётом архитектурных ограничений.

**КОД (РЕАЛИЗАЦИЯ):**
```php
<?php
/**
 * Data Adapter - подготовка данных для UI
 * ТЗ-3: Полная реализация
 */

if (!defined('ABSPATH')) {
    exit;
}

final class NPK_Concentration_Data_Adapter {
    
    private static $instance = null;
    
    /**
     * Приватный конструктор
     */
    private function __construct() {}
    
    /**
     * Получить экземпляр класса (синглтон)
     * @return self
     */
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    /**
     * Получить данные для инициализации UI согласно ТЗ п.7.3.1
     * @return array Структурированные данные для JavaScript
     */
    public function get_ui_data(): array {
        // Проверка существования класса базы данных
        if (!class_exists('NPK_Database')) {
            return $this->get_empty_ui_data();
        }
        
        // 1. Получаем данные из БД
        $fertilizers = $this->prepare_fertilizers();
        $profiles = $this->prepare_profiles();
        $constants = NPK_Database::get_constants();
        
        // 2. Определяем профиль по умолчанию (влажный)
        $default_ground_type = 'wet';
        foreach ($profiles as $type => $profile) {
            if (isset($profile['is_default']) && $profile['is_default']) {
                $default_ground_type = $type;
                break;
            }
        }
        
        // 3. Определяем удобрение по умолчанию (первое по ID)
        $default_fertilizer = null;
        if (!empty($fertilizers)) {
            $default_fertilizer = reset($fertilizers);
        }
        
        // 4. Формируем структуру согласно ТЗ п.7.3.1
        return [
            // Конфигурация
            'constants' => $constants,
            
            // Справочники
            'fertilizers' => $fertilizers,
            'profiles' => $profiles,
            
            // Начальные значения UI согласно ТЗ п.6.1
            'defaults' => [
                'fertilizer_id' => $default_fertilizer ? $default_fertilizer['id'] : 0,
                'ground_type' => $default_ground_type,
                'concentration' => 0.3,
                'water_volume' => 10.0
            ],
            
            // AJAX endpoint
            'ajax' => [
                'url' => admin_url('admin-ajax.php'),
                'nonce' => wp_create_nonce('npk_concentration_calculate')
            ]
        ];
    }
    
    /**
     * Подготовить удобрения для UI согласно ТЗ п.7.3.1
     * @return array Массив удобрений, отсортированный по id (возрастание)
     */
    private function prepare_fertilizers(): array {
        $raw_fertilizers = NPK_Database::get_fertilizers();
        $prepared = [];
        
        foreach ($raw_fertilizers as $fertilizer) {
            // Валидация растворимости для отображения
            $solubility_total = $this->validate_solubility_for_display($fertilizer['solubility_total']);
            
            // Вычисляем сумму NPK, если поле отсутствует
            $npk_sum = $fertilizer['npk_sum'] ?? array_sum($fertilizer['npk']);
            
            $prepared[] = [
                'id' => (int) $fertilizer['id'],
                'name' => (string) $fertilizer['name'],
                'npk_array' => array_map('intval', $fertilizer['npk']),
                'npk_sum' => (float) $npk_sum,
                'solubility_total' => (float) $solubility_total,
                'type' => (string) $fertilizer['type'],
                'notes' => (string) ($fertilizer['notes'] ?? '')
            ];
        }
        
        // Сортировка по возрастанию id согласно ТЗ п.5
        usort($prepared, function($a, $b) {
            return $a['id'] <=> $b['id'];
        });
        
        return $prepared;
    }
    
    /**
     * Подготовить профили грунтов для UI
     * @return array Ассоциативный массив профилей
     */
    private function prepare_profiles(): array {
        $raw_profiles = NPK_Database::get_concentration_profiles();
        $prepared = [];
        
        foreach ($raw_profiles as $type => $profile) {
            $prepared[$type] = [
                'name' => (string) $profile['name'],
                'emoji' => (string) $profile['emoji'],
                'description' => (string) $profile['description'],
                'is_default' => (bool) ($profile['is_default'] ?? false),
                'states' => $this->prepare_states($profile['states'])
            ];
        }
        
        return $prepared;
    }
    
    /**
     * Подготовить состояния для профиля
     * @param array $states Массив состояний из БД
     * @return array Подготовленные состояния
     */
    private function prepare_states(array $states): array {
        $prepared = [];
        
        foreach ($states as $state) {
            $prepared[] = [
                'id' => (string) $state['id'],
                'name' => (string) $state['name'],
                'color' => (string) $state['color'],
                'emoji' => (string) $state['emoji'],
                'description' => (string) $state['description'],
                'range' => [
                    (float) $state['range'][0],
                    (float) $state['range'][1]
                ]
            ];
        }
        
        return $prepared;
    }
    
    /**
     * Валидация растворимости для отображения
     * Отдельная логика от Calculation_Engine, только для UI
     * @param mixed $value Значение растворимости
     * @return float Валидированное значение для отображения
     */
    private function validate_solubility_for_display($value): float {
        $float_value = (float) $value;
        
        if ($float_value <= 0 || $float_value > 100) {
            return 100.0; // Максимальное значение для отображения
        }
        
        return $float_value;
    }
    
    /**
     * Получить удобрение по ID (дополнительный метод для гибкости)
     * @param int $fertilizer_id ID удобрения
     * @return array|null Данные удобрения или null
     */
    public function get_fertilizer_by_id(int $fertilizer_id): ?array {
        $fertilizers = $this->prepare_fertilizers();
        
        foreach ($fertilizers as $fertilizer) {
            if ($fertilizer['id'] === $fertilizer_id) {
                return $fertilizer;
            }
        }
        
        return null;
    }
    
    /**
     * Получить профиль грунта по типу (дополнительный метод для гибкости)
     * @param string $ground_type Тип грунта
     * @return array|null Данные профиля или null
     */
    public function get_ground_profile(string $ground_type): ?array {
        $profiles = $this->prepare_profiles();
        
        return $profiles[$ground_type] ?? null;
    }
    
    /**
     * Получить пустую структуру UI данных при ошибке
     * @return array Безопасная пустая структура
     */
    private function get_empty_ui_data(): array {
        return [
            'constants' => ['thresholds' => ['water_volume_max' => 200]],
            'fertilizers' => [],
            'profiles' => [],
            'defaults' => [
                'fertilizer_id' => 0,
                'ground_type' => 'wet',
                'concentration' => 0.3,
                'water_volume' => 10.0
            ],
            'ajax' => [
                'url' => admin_url('admin-ajax.php'),
                'nonce' => wp_create_nonce('npk_concentration_calculate')
            ]
        ];
    }
    
    // Запрещаем клонирование и восстановление из сериализации
    private function __clone() {}
    public function __wakeup() {}
}
```
