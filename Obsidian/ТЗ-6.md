/**
 * NPK Concentration Calculator - UI логика
 * ТЗ-6: Полная реализация
 */

(function() {
    'use strict';
    
    // ==================== КОНСТАНТЫ И КОНФИГУРАЦИЯ ====================
    const DEFAULT_DEBOUNCE_DELAY = 500; // 500мс согласно ТЗ п.6.2
    const AJAX_TIMEOUT = 10000; // 10 секунд таймаут
    const RETRY_DELAY = 5000; // 5 секунд согласно ТЗ п.7.4
    
    // ==================== КЛАСС КАЛЬКУЛЯТОРА ====================
    class NPKConcentrationCalculator {
        
        /**
         * Конструктор калькулятора
         * @param {HTMLElement} container - Корневой элемент калькулятора
         */
        constructor(container) {
            this.container = container;
            this.calculatorId = container.dataset.npkCalculatorId;
            this.data = this.getCalculatorData();
            
            // Состояние калькулятора
            this.state = {
                fertilizerId: this.data.defaults.fertilizer_id || 0,
                groundType: this.data.defaults.ground_type || 'wet',
                concentration: this.data.defaults.concentration || 0.3,
                waterVolume: this.data.defaults.water_volume || 10.0,
                lastServerResponse: null,
                isCalculating: false,
                hasError: false
            };
            
            // Контроллер для отмены AJAX запросов
            this.abortController = null;
            
            // Таймеры
            this.debounceTimer = null;
            this.retryTimer = null;
            
            // Инициализация
            this.initElements();
            this.initEventListeners();
            this.initUI();
            this.performInitialCalculation();
            
            console.log(`NPK Calculator initialized: ${this.calculatorId}`);
        }
        
        /**
         * Получение данных для калькулятора из window
         * @returns {Object} Данные калькулятора
         */
        getCalculatorData() {
            const varName = `NPK_Concentration_Data_${this.calculatorId.replace(/-/g, '_')}`;
            return window[varName] || window.NPK_Concentration_Instances?.[this.calculatorId] || {};
        }
        
        /**
         * Инициализация DOM элементов
         */
        initElements() {
            // Грунт
            this.groundButtons = {
                dry: this.container.querySelector('[data-npk-ground-type="dry"]'),
                medium: this.container.querySelector('[data-npk-ground-type="medium"]'),
                wet: this.container.querySelector('[data-npk-ground-type="wet"]')
            };
            
            // Удобрения
            this.fertilizerSelect = this.container.querySelector('[data-npk-fertilizer-select]');
            this.npkDisplay = this.container.querySelector('[data-npk-fertilizer-npk]');
            this.solubilityDisplay = this.container.querySelector('[data-npk-fertilizer-solubility] .npk-solubility-value');
            this.notesDisplay = this.container.querySelector('[data-npk-fertilizer-notes]');
            
            // Поля ввода
            this.waterVolumeInput = this.container.querySelector('[data-npk-input="water-volume"]');
            this.concentrationInput = this.container.querySelector('[data-npk-input="concentration"]');
            this.npkConcentrationDisplay = this.container.querySelector('[data-npk-npk-concentration] .npk-npk-concentration-value');
            
            // Кнопки +/-
            this.plusMinusButtons = this.container.querySelectorAll('[data-npk-action]');
            
            // Слайдер
            this.slider = this.container.querySelector('[data-npk-slider]');
            this.sliderTrack = this.container.querySelector('[data-npk-slider-track]');
            this.sliderFill = this.container.querySelector('[data-npk-slider-fill]');
            this.stateLabel = this.container.querySelector('[data-npk-state-label]');
            this.stateName = this.container.querySelector('[data-npk-state-name]');
            
            // Результаты
            this.massResult = this.container.querySelector('[data-npk-result-mass] .npk-mass-value');
            this.concentrationResult = this.container.querySelector('[data-npk-result-concentration] .npk-concentration-value');
            this.stateResult = this.container.querySelector('[data-npk-result-state]');
            this.stateEmoji = this.container.querySelector('[data-npk-state-emoji]');
            this.stateResultName = this.container.querySelector('[data-npk-result-state] [data-npk-state-name]');
            this.warningsContainer = this.container.querySelector('[data-npk-warnings]');
            this.stateDescription = this.container.querySelector('[data-npk-state-description]');
            
            // Статус
            this.statusIndicator = this.container.querySelector('.npk-status-indicator');
            this.statusText = this.container.querySelector('.npk-status-text');
        }
        
        /**
         * Инициализация обработчиков событий
         */
        initEventListeners() {
            // Кнопки выбора грунта
            Object.values(this.groundButtons).forEach(button => {
                button.addEventListener('click', (e) => this.handleGroundChange(e));
            });
            
            // Выбор удобрения
            this.fertilizerSelect.addEventListener('change', (e) => this.handleFertilizerChange(e));
            
            // Поля ввода с дебаунсом
            this.waterVolumeInput.addEventListener('input', 
                this.debounce(() => this.handleWaterVolumeChange(), DEFAULT_DEBOUNCE_DELAY)
            );
            this.waterVolumeInput.addEventListener('blur', 
                () => this.handleWaterVolumeBlur()
            );
            
            this.concentrationInput.addEventListener('input', 
                this.debounce(() => this.handleConcentrationChange(), DEFAULT_DEBOUNCE_DELAY)
            );
            this.concentrationInput.addEventListener('blur', 
                () => this.handleConcentrationBlur()
            );
            
            // Кнопки +/-
            this.plusMinusButtons.forEach(button => {
                button.addEventListener('click', (e) => this.handlePlusMinusClick(e));
                button.addEventListener('mousedown', (e) => this.handlePlusMinusMouseDown(e));
                button.addEventListener('mouseup', () => this.handlePlusMinusMouseUp());
                button.addEventListener('mouseleave', () => this.handlePlusMinusMouseUp());
            });
            
            // Слайдер
            this.slider.addEventListener('input', 
                this.debounce(() => this.handleSliderChange(), DEFAULT_DEBOUNCE_DELAY)
            );
            this.slider.addEventListener('change', () => this.handleSliderChangeEnd());
        }
        
        /**
         * Инициализация UI начальными данными
         */
        initUI() {
            // Заполняем список удобрений
            this.populateFertilizerSelect();
            
            // Устанавливаем начальные значения
            this.updateGroundButtons(this.state.groundType);
            this.updateFertilizerInfo(this.state.fertilizerId);
            this.updateInputValues();
            this.updateSlider();
            this.updateNPKConcentration();
            
            // Обновляем доступность кнопок
            this.updateButtonStates();
        }
        
        /**
         * Заполнение выпадающего списка удобрений
         */
        populateFertilizerSelect() {
            if (!this.data.fertilizers || !Array.isArray(this.data.fertilizers)) {
                console.error('No fertilizer data available');
                return;
            }
            
            // Очищаем список
            this.fertilizerSelect.innerHTML = '';
            
            // Добавляем удобрения
            this.data.fertilizers.forEach(fertilizer => {
                const option = document.createElement('option');
                option.value = fertilizer.id;
                option.textContent = `${fertilizer.name} [${fertilizer.npk_array.join('-')}]`;
                
                if (fertilizer.id === this.state.fertilizerId) {
                    option.selected = true;
                }
                
                this.fertilizerSelect.appendChild(option);
            });
        }
        
        /**
         * Выполнение начального расчёта при загрузке
         */
        performInitialCalculation() {
            // Сначала локальный расчёт для мгновенного отображения
            this.performLocalCalculation();
            
            // Затем серверный расчёт для точных данных
            this.performServerCalculation();
        }
        
        // ==================== ОБРАБОТЧИКИ СОБЫТИЙ ====================
        
        /**
         * Обработка изменения грунта
         * @param {Event} e - Событие клика
         */
        handleGroundChange(e) {
            const groundType = e.currentTarget.dataset.npkGroundType;
            
            requestAnimationFrame(() => {
                this.state.groundType = groundType;
                this.updateGroundButtons(groundType);
                this.performServerCalculation(); // Немедленный запрос согласно ТЗ п.6.2
            });
        }
        
        /**
         * Обработка изменения удобрения
         * @param {Event} e - Событие change
         */
        handleFertilizerChange(e) {
            const fertilizerId = parseInt(e.target.value, 10);
            
            requestAnimationFrame(() => {
                this.state.fertilizerId = fertilizerId;
                this.updateFertilizerInfo(fertilizerId);
                this.updateNPKConcentration();
                this.performServerCalculation(); // Немедленный запрос согласно ТЗ п.6.2
            });
        }
        
        /**
         * Обработка изменения объёма воды
         */
        handleWaterVolumeChange() {
            const value = this.validateWaterVolume(this.waterVolumeInput.value);
            
            requestAnimationFrame(() => {
                this.state.waterVolume = value;
                this.waterVolumeInput.value = value.toFixed(1);
                this.performLocalCalculation();
                this.scheduleServerCalculation();
            });
        }
        
        /**
         * Обработка потери фокуса для объёма воды
         */
        handleWaterVolumeBlur() {
            const value = this.validateWaterVolume(this.waterVolumeInput.value);
            
            requestAnimationFrame(() => {
                this.state.waterVolume = value;
                this.waterVolumeInput.value = value.toFixed(1);
                this.updateButtonStates();
                this.performServerCalculation();
            });
        }
        
        /**
         * Обработка изменения концентрации
         */
        handleConcentrationChange() {
            const value = this.validateConcentration(this.concentrationInput.value);
            
            requestAnimationFrame(() => {
                this.state.concentration = value;
                this.concentrationInput.value = value.toFixed(2);
                this.updateSliderValue(value);
                this.updateConcentrationColor();
                this.updateNPKConcentration();
                this.performLocalCalculation();
                this.scheduleServerCalculation();
            });
        }
        
        /**
         * Обработка потери фокуса для концентрации
         */
        handleConcentrationBlur() {
            const value = this.validateConcentration(this.concentrationInput.value);
            
            requestAnimationFrame(() => {
                this.state.concentration = value;
                this.concentrationInput.value = value.toFixed(2);
                this.updateSliderValue(value);
                this.updateButtonStates();
                this.performServerCalculation();
            });
        }
        
        /**
         * Обработка кликов на кнопки +/-
         * @param {Event} e - Событие клика
         */
        handlePlusMinusClick(e) {
            const button = e.currentTarget;
            const inputType = button.dataset.npkInput;
            const action = button.dataset.npkAction;
            
            this.adjustValue(inputType, action);
        }
        
        /**
         * Обработка зажатия кнопки +/-
         * @param {Event} e - Событие mousedown
         */
        handlePlusMinusMouseDown(e) {
            const button = e.currentTarget;
            const inputType = button.dataset.npkInput;
            const action = button.dataset.npkAction;
            
            // Начинаем повторное изменение при зажатии
            this.repeatInterval = setInterval(() => {
                this.adjustValue(inputType, action);
            }, 150); // 150мс для плавного повторения
            
            // Первое изменение сразу
            this.adjustValue(inputType, action);
        }
        
        /**
         * Обработка отпускания кнопки +/-
         */
        handlePlusMinusMouseUp() {
            if (this.repeatInterval) {
                clearInterval(this.repeatInterval);
                this.repeatInterval = null;
            }
        }
        
        /**
         * Обработка изменения слайдера
         */
        handleSliderChange() {
            const value = parseFloat(this.slider.value);
            
            requestAnimationFrame(() => {
                this.state.concentration = value;
                this.concentrationInput.value = value.toFixed(2);
                this.updateConcentrationColor();
                this.updateNPKConcentration();
                this.performLocalCalculation();
                // Для слайдера - только предварительный расчёт согласно ТЗ п.6.2
            });
        }
        
        /**
         * Обработка окончания изменения слайдера
         */
        handleSliderChangeEnd() {
            const value = parseFloat(this.slider.value);
            
            requestAnimationFrame(() => {
                this.state.concentration = value;
                this.concentrationInput.value = value.toFixed(2);
                this.performServerCalculation(); // Точный расчёт после отпускания
            });
        }
        
        // ==================== РАСЧЁТЫ ====================
        
        /**
         * Локальный расчёт для мгновенного обновления UI
         */
        performLocalCalculation() {
            // Получаем предварительное состояние согласно ТЗ п.6.2
            const previewState = this.getPreviewState();
            
            // Упрощённый расчёт массы для UX
            const previewMass = this.calculatePreviewMass();
            
            // Обновляем UI предварительными данными
            requestAnimationFrame(() => {
                this.updateResults(previewMass, this.state.concentration, previewState);
                this.updateStateDisplay(previewState);
                this.updateConcentrationColor(previewState);
            });
        }
        
        /**
         * Серверный расчёт через AJAX
         */
        async performServerCalculation() {
            // Отменяем предыдущий запрос если есть
            if (this.abortController) {
                this.abortController.abort();
            }
            
            // Создаём новый контроллер
            this.abortController = new AbortController();
            
            // Обновляем статус
            this.setCalculating(true);
            
            try {
                // Формируем запрос согласно ТЗ п.6.6.1
                const response = await fetch(this.data.ajax.url, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-WP-Nonce': this.data.ajax.nonce
                    },
                    body: JSON.stringify({
                        action: 'npk_concentration_calculate',
                        fertilizer_id: this.state.fertilizerId,
                        concentration: this.state.concentration,
                        water_volume: this.state.waterVolume,
                        ground_type: this.state.groundType
                    }),
                    signal: this.abortController.signal
                });
                
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                
                const result = await response.json();
                
                if (result.success) {
                    this.handleServerSuccess(result.data);
                } else {
                    this.handleServerError(result.data?.message || 'Ошибка сервера');
                }
                
            } catch (error) {
                if (error.name === 'AbortError') {
                    // Запрос был отменён - ничего не делаем
                    return;
                }
                
                this.handleServerError('Ошибка сети или сервера');
                this.scheduleRetry();
                
            } finally {
                this.setCalculating(false);
                this.abortController = null;
            }
        }
        
        /**
         * Планирование серверного расчёта с дебаунсом
         */
        scheduleServerCalculation() {
            if (this.debounceTimer) {
                clearTimeout(this.debounceTimer);
            }
            
            this.debounceTimer = setTimeout(() => {
                this.performServerCalculation();
            }, DEFAULT_DEBOUNCE_DELAY);
        }
        
        /**
         * Планирование повторной попытки при ошибке
         */
        scheduleRetry() {
            if (this.retryTimer) {
                clearTimeout(this.retryTimer);
            }
            
            this.retryTimer = setTimeout(() => {
                this.performServerCalculation();
            }, RETRY_DELAY);
        }
        
        // ==================== ВСПОМОГАТЕЛЬНЫЕ МЕТОДЫ ====================
        
        /**
         * Получение предварительного состояния для UX
         * @returns {Object} Предварительные данные состояния
         */
        getPreviewState() {
            if (!this.data.profiles || !this.data.profiles[this.state.groundType]) {
                return this.getDefaultState();
            }
            
            const profile = this.data.profiles[this.state.groundType];
            const states = profile.states;
            const concentration = this.state.concentration;
            
            // Алгоритм определения состояния согласно ТЗ п.4.3
            for (let i = 0; i < states.length; i++) {
                const state = states[i];
                const [min, max] = state.range;
                
                // Последнее состояние - включительно
                if (i === states.length - 1) {
                    if (concentration >= min && concentration <= max) {
                        return state;
                    }
                } 
                // Остальные состояния - полуоткрытый интервал
                else if (concentration >= min && concentration < max) {
                    return state;
                }
            }
            
            // Если не нашли - возвращаем последнее состояние
            return states[states.length - 1] || this.getDefaultState();
        }
        
        /**
         * Упрощённый расчёт массы для UX
         * @returns {number} Предварительная масса
         */
        calculatePreviewMass() {
            // Базовая формула без учёта растворимости для UX
            const baseMass = this.state.concentration * this.state.waterVolume * 10; // *1000/100
            
            // Округляем до целых
            return Math.round(baseMass);
        }
        
        /**
         * Обновление результатов в UI
         * @param {number} mass - Масса удобрения
         * @param {number} concentration - Концентрация
         * @param {Object} state - Состояние
         */
        updateResults(mass, concentration, state) {
            requestAnimationFrame(() => {
                // Масса
                if (this.massResult) {
                    this.massResult.textContent = mass;
                }
                
                // Концентрация
                if (this.concentrationResult) {
                    this.concentrationResult.textContent = concentration.toFixed(2);
                }
                
                // Состояние
                if (state) {
                    this.updateStateDisplay(state);
                }
            });
        }
        
        /**
         * Обновление отображения состояния
         * @param {Object} state - Данные состояния
         */
        updateStateDisplay(state) {
            requestAnimationFrame(() => {
                // Название состояния на метке и в результатах
                if (this.stateName) {
                    this.stateName.textContent = state.name;
                }
                if (this.stateResultName) {
                    this.stateResultName.textContent = state.name;
                }
                
                // Эмодзи состояния
                if (this.stateEmoji) {
                    this.stateEmoji.textContent = state.emoji;
                }
                
                // Описание состояния
                if (this.stateDescription) {
                    this.stateDescription.textContent = state.description;
                }
                
                // Цвет состояния
                this.updateElementColors(state.color);
            });
        }
        
        /**
         * Обновление цветов элементов согласно состоянию
         * @param {string} color - Цвет состояния
         */
        updateElementColors(color) {
            requestAnimationFrame(() => {
                // Метка над слайдером
                if (this.stateLabel) {
                    this.stateLabel.style.backgroundColor = color;
                    this.stateLabel.style.color = this.getContrastColor(color);
                }
                
                // Заливка слайдера
                if (this.sliderFill) {
                    this.sliderFill.style.backgroundColor = color;
                }
                
                // Плашка состояния в результатах
                if (this.stateResult) {
                    this.stateResult.style.backgroundColor = color;
                    this.stateResult.style.color = this.getContrastColor(color);
                }
            });
        }
        
        /**
         * Обновление цвета фона поля концентрации
         * @param {Object} state - Текущее состояние (опционально)
         */
        updateConcentrationColor(state = null) {
            if (!state) {
                state = this.getPreviewState();
            }
            
            const [min, max] = state.range;
            const concentration = this.state.concentration;
            
            // Рассчитываем вес для интерполяции
            const rangeWidth = max - min;
            const weight = rangeWidth > 0 ? (concentration - min) / rangeWidth : 0;
            
            // Интерполяция цвета от белого к цвету состояния
            const interpolatedColor = this.interpolateColor('#FFFFFF', state.color, weight);
            
            requestAnimationFrame(() => {
                if (this.concentrationInput) {
                    this.concentrationInput.style.backgroundColor = interpolatedColor;
                }
            });
        }
        
        /**
         * Интерполяция цвета между двумя значениями
         * @param {string} color1 - Начальный цвет
         * @param {string} color2 - Конечный цвет
         * @param {number} weight - Вес (0-1)
         * @returns {string} Интерполированный цвет
         */
        interpolateColor(color1, color2, weight) {
            const hex = color => color.replace('#', '');
            const toRGB = hex => [
                parseInt(hex.substr(0, 2), 16),
                parseInt(hex.substr(2, 2), 16),
                parseInt(hex.substr(4, 2), 16)
            ];
            
            const rgb1 = toRGB(hex(color1));
            const rgb2 = toRGB(hex(color2));
            
            const result = rgb1.map((channel, i) => {
                return Math.round(channel + (rgb2[i] - channel) * weight);
            });
            
            return `rgb(${result.join(', ')})`;
        }
        
        /**
         * Получение контрастного цвета (чёрный/белый)
         * @param {string} hexColor - HEX цвет
         * @returns {string} Контрастный цвет
         */
        getContrastColor(hexColor) {
            const hex = hexColor.replace('#', '');
            const r = parseInt(hex.substr(0, 2), 16);
            const g = parseInt(hex.substr(2, 2), 16);
            const b = parseInt(hex.substr(4, 2), 16);
            
            // Формула яркости
            const brightness = (r * 299 + g * 587 + b * 114) / 1000;
            
            return brightness > 128 ? '#000000' : '#FFFFFF';
        }
        
        // ==================== ВАЛИДАЦИЯ И КОРРЕКЦИЯ ====================
        
        /**
         * Валидация концентрации
         * @param {string|number} value - Значение концентрации
         * @returns {number} Валидированное значение
         */
        validateConcentration(value) {
            let numValue = parseFloat(value.toString().replace(',', '.'));
            
            if (isNaN(numValue)) {
                return this.state.concentration;
            }
            
            // Коррекция границ согласно ТЗ п.6.3
            numValue = Math.max(0.05, Math.min(1.5, numValue));
            
            // Округление до двух знаков
            return Math.round(numValue * 100) / 100;
        }
        
        /**
         * Валидация объёма воды
         * @param {string|number} value - Значение объёма
         * @returns {number} Валидированное значение
         */
        validateWaterVolume(value) {
            let numValue = parseFloat(value.toString().replace(',', '.'));
            
            if (isNaN(numValue)) {
                return this.state.waterVolume;
            }
            
            // Коррекция границ
            const min = this.data.constants?.thresholds?.water_volume_min || 0.5;
            const max = this.data.constants?.thresholds?.water_volume_max || 200;
            numValue = Math.max(min, Math.min(max, numValue));
            
            // Округление до одного знака
            return Math.round(numValue * 10) / 10;
        }
        
        /**
         * Корректировка значения через кнопки +/-
         * @param {string} inputType - Тип поля
         * @param {string} action - Действие (increase/decrease)
         */
        adjustValue(inputType, action) {
            let value, step;
            
            if (inputType === 'water-volume') {
                value = this.state.waterVolume;
                // Шаг зависит от значения согласно ТЗ п.6.3
                step = value < 10 ? 0.5 : 1.0;
            } else if (inputType === 'concentration') {
                value = this.state.concentration;
                step = 0.01;
            } else {
                return;
            }
            
            // Изменение значения
            if (action === 'increase') {
                value += step;
            } else if (action === 'decrease') {
                value -= step;
            }
            
            // Валидация и обновление
            if (inputType === 'water-volume') {
                value = this.validateWaterVolume(value);
                this.state.waterVolume = value;
                this.waterVolumeInput.value = value.toFixed(1);
                this.performLocalCalculation();
                this.scheduleServerCalculation();
            } else if (inputType === 'concentration') {
                value = this.validateConcentration(value);
                this.state.concentration = value;
                this.concentrationInput.value = value.toFixed(2);
                this.updateSliderValue(value);
                this.updateConcentrationColor();
                this.updateNPKConcentration();
                this.performLocalCalculation();
                this.scheduleServerCalculation();
            }
            
            // Обновление состояния кнопок
            this.updateButtonStates();
        }
        
        // ==================== ОБНОВЛЕНИЕ UI ====================
        
        /**
         * Обновление кнопок выбора грунта
         * @param {string} activeType - Активный тип грунта
         */
        updateGroundButtons(activeType) {
            Object.entries(this.groundButtons).forEach(([type, button]) => {
                requestAnimationFrame(() => {
                    if (type === activeType) {
                        button.classList.add('npk-ground-active');
                        button.setAttribute('aria-pressed', 'true');
                    } else {
                        button.classList.remove('npk-ground-active');
                        button.setAttribute('aria-pressed', 'false');
                    }
                });
            });
        }
        
        /**
         * Обновление информации об удобрении
         * @param {number} fertilizerId - ID удобрения
         */
        updateFertilizerInfo(fertilizerId) {
            const fertilizer = this.data.fertilizers?.find(f => f.id === fertilizerId);
            
            if (!fertilizer) return;
            
            requestAnimationFrame(() => {
                // NPK состав
                if (this.npkDisplay) {
                    const npkElements = this.npkDisplay.querySelectorAll('[data-npk-n], [data-npk-p], [data-npk-k]');
                    if (npkElements[0]) npkElements[0].textContent = fertilizer.npk_array[0];
                    if (npkElements[1]) npkElements[1].textContent = fertilizer.npk_array[1];
                    if (npkElements[2]) npkElements[2].textContent = fertilizer.npk_array[2];
                }
                
                // Растворимость
                if (this.solubilityDisplay) {
                    this.solubilityDisplay.textContent = fertilizer.solubility_total;
                }
                
                // Примечания
                if (this.notesDisplay && fertilizer.notes) {
                    this.notesDisplay.textContent = fertilizer.notes;
                }
            });
        }
        
        /**
         * Обновление концентрации NPK
         */
        updateNPKConcentration() {
            const fertilizer = this.data.fertilizers?.find(f => f.id === this.state.fertilizerId);
            
            if (!fertilizer || !fertilizer.npk_sum) return;
            
            // Формула согласно ТЗ п.6.3: общая_концентрация × (сумма_NPK / 100)
            const npkConcentration = this.state.concentration * (fertilizer.npk_sum / 100);
            
            requestAnimationFrame(() => {
                if (this.npkConcentrationDisplay) {
                    this.npkConcentrationDisplay.textContent = npkConcentration.toFixed(3);
                }
            });
        }
        
        /**
         * Обновление значений полей ввода
         */
        updateInputValues() {
            requestAnimationFrame(() => {
                if (this.waterVolumeInput) {
                    this.waterVolumeInput.value = this.state.waterVolume.toFixed(1);
                }
                if (this.concentrationInput) {
                    this.concentrationInput.value = this.state.concentration.toFixed(2);
                }
            });
        }
        
        /**
         * Обновление слайдера
         */
        updateSlider() {
            requestAnimationFrame(() => {
                if (this.slider) {
                    this.slider.value = this.state.concentration;
                    this.updateSliderFill();
                }
            });
        }
        
        /**
         * Обновление значения слайдера
         * @param {number} value - Значение концентрации
         */
        updateSliderValue(value) {
            requestAnimationFrame(() => {
                if (this.slider) {
                    this.slider.value = value;
                    this.updateSliderFill();
                }
            });
        }
        
        /**
         * Обновление заливки слайдера
         */
        updateSliderFill() {
            if (!this.slider || !this.sliderFill) return;
            
            const value = parseFloat(this.slider.value);
            const min = parseFloat(this.slider.min);
            const max = parseFloat(this.slider.max);
            
            const percentage = ((value - min) / (max - min)) * 100;
            
            requestAnimationFrame(() => {
                this.sliderFill.style.width = `${percentage}%`;
            });
        }
        
        /**
         * Обновление состояния кнопок +/-
         */
        updateButtonStates() {
            requestAnimationFrame(() => {
                // Кнопки объёма воды
                const waterMin = this.data.constants?.thresholds?.water_volume_min || 0.5;
                const waterMax = this.data.constants?.thresholds?.water_volume_max || 200;
                
                const waterMinusButton = this.container.querySelector('[data-npk-input="water-volume"][data-npk-action="decrease"]');
                const waterPlusButton = this.container.querySelector('[data-npk-input="water-volume"][data-npk-action="increase"]');
                
                if (waterMinusButton) {
                    waterMinusButton.disabled = this.state.waterVolume <= waterMin;
                }
                if (waterPlusButton) {
                    waterPlusButton.disabled = this.state.waterVolume >= waterMax;
                }
                
                // Кнопки концентрации
                const concMin = 0.05;
                const concMax = 1.5;
                
                const concMinusButton = this.container.querySelector('[data-npk-input="concentration"][data-npk-action="decrease"]');
                const concPlusButton = this.container.querySelector('[data-npk-input="concentration"][data-npk-action="increase"]');
                
                if (concMinusButton) {
                    concMinusButton.disabled = this.state.concentration <= concMin;
                }
                if (concPlusButton) {
                    concPlusButton.disabled = this.state.concentration >= concMax;
                }
            });
        }
        
        /**
         * Установка состояния "расчёт"
         * @param {boolean} isCalculating - Флаг расчёта
         */
        setCalculating(isCalculating) {
            this.state.isCalculating = isCalculating;
            
            requestAnimationFrame(() => {
                if (isCalculating) {
                    this.container.classList.add('npk-calculating');
                    if (this.statusIndicator) {
                        this.statusIndicator.classList.remove('npk-status-ready', 'npk-status-error');
                        this.statusIndicator.classList.add('npk-status-calculating');
                    }
                    if (this.statusText) {
                        this.statusText.textContent = 'Расчёт...';
                    }
                } else {
                    this.container.classList.remove('npk-calculating');
                    if (this.statusIndicator && !this.state.hasError) {
                        this.statusIndicator.classList.remove('npk-status-calculating', 'npk-status-error');
                        this.statusIndicator.classList.add('npk-status-ready');
                    }
                    if (this.statusText && !this.state.hasError) {
                        this.statusText.textContent = 'Готов к работе';
                    }
                }
            });
        }
        
        /**
         * Обработка успешного ответа сервера
         * @param {Object} result - Результат расчёта
         */
        handleServerSuccess(result) {
            this.state.lastServerResponse = result;
            this.state.hasError = false;
            
            // Обновляем UI точными данными с сервера
            requestAnimationFrame(() => {
                // Масса
                if (this.massResult) {
                    this.massResult.textContent = result.fertilizer_mass;
                }
                
                // Концентрация (может быть скорректирована сервером)
                if (this.concentrationResult) {
                    this.concentrationResult.textContent = result.concentration.toFixed(2);
                }
                if (this.concentrationInput && result.concentration !== this.state.concentration) {
                    this.state.concentration = result.concentration;
                    this.concentrationInput.value = result.concentration.toFixed(2);
                    this.updateSliderValue(result.concentration);
                }
                
                // Объём воды (может быть скорректирован сервером)
                if (this.waterVolumeInput && result.water_volume !== this.state.waterVolume) {
                    this.state.waterVolume = result.water_volume;
                    this.waterVolumeInput.value = result.water_volume.toFixed(1);
                }
                
                // Состояние
                if (result.state) {
                    this.updateStateDisplay(result.state);
                    this.updateConcentrationColor(result.state);
                }
                
                // Предупреждения
                this.updateWarnings(result.warnings);
                
                // Концентрация NPK
                this.updateNPKConcentration();
                
                // Обновляем состояние кнопок
                this.updateButtonStates();
            });
        }
        
        /**
         * Обработка ошибки сервера
         * @param {string} message - Сообщение об ошибке
         */
        handleServerError(message) {
            this.state.hasError = true;
            
            requestAnimationFrame(() => {
                // Показываем уведомление согласно ТЗ п.7.4
                this.showWarning(message);
                
                // Обновляем статус
                if (this.statusIndicator) {
                    this.statusIndicator.classList.remove('npk-status-ready', 'npk-status-calculating');
                    this.statusIndicator.classList.add('npk-status-error');
                }
                if (this.statusText) {
                    this.statusText.textContent = 'Ошибка соединения';
                }
                
                // Продолжаем работать на локальных расчётах
                this.performLocalCalculation();
            });
        }
        
        /**
         * Обновление предупреждений
         * @param {Array} warnings - Массив предупреждений
         */
        updateWarnings(warnings) {
            if (!this.warningsContainer) return;
            
            requestAnimationFrame(() => {
                this.warningsContainer.innerHTML = '';
                
                if (warnings && warnings.length > 0) {
                    warnings.forEach(warning => {
                        const warningElement = document.createElement('div');
                        warningElement.className = 'npk-warning';
                        warningElement.textContent = warning;
                        this.warningsContainer.appendChild(warningElement);
                    });
                    
                    this.warningsContainer.classList.add('has-warnings');
                } else {
                    this.warningsContainer.classList.remove('has-warnings');
                }
            });
        }
        
        /**
         * Показ предупреждения
         * @param {string} message - Сообщение
         */
        showWarning(message) {
            if (!this.warningsContainer) return;
            
            requestAnimationFrame(() => {
                const warningElement = document.createElement('div');
                warningElement.className = 'npk-warning npk-warning-error';
                warningElement.textContent = message;
                
                this.warningsContainer.innerHTML = '';
                this.warningsContainer.appendChild(warningElement);
                this.warningsContainer.classList.add('has-warnings');
                
                // Автоматическое скрытие через 5 секунд
                setTimeout(() => {
                    if (warningElement.parentNode === this.warningsContainer) {
                        warningElement.remove();
                        if (this.warningsContainer.children.length === 0) {
                            this.warningsContainer.classList.remove('has-warnings');
                        }
                    }
                }, 5000);
            });
        }
        
        /**
         * Получение состояния по умолчанию
         * @returns {Object} Состояние по умолчанию
         */
        getDefaultState() {
            return {
                id: 'unknown',
                name: 'Неизвестно',
                color: '#6c757d',
                emoji: '❓',
                description: 'Не удалось определить состояние',
                range: [0, 0]
            };
        }
        
        /**
         * Функция дебаунса
         * @param {Function} func - Функция для выполнения
         * @param {number} wait - Время ожидания
         * @returns {Function} Обёрнутая функция
         */
        debounce(func, wait) {
            return (...args) => {
                if (this.debounceTimer) {
                    clearTimeout(this.debounceTimer);
                }
                this.debounceTimer = setTimeout(() => func.apply(this, args), wait);
            };
        }
    }
    
    // ==================== ИНИЦИАЛИЗАЦИЯ ВСЕХ КАЛЬКУЛЯТОРОВ ====================
    
    /**
     * Инициализация всех калькуляторов на странице
     */
    function initCalculators() {
        const calculators = document.querySelectorAll('.npk-concentration-calculator');
        
        if (calculators.length === 0) {
            return;
        }
        
        // Инициализируем каждый калькулятор
        calculators.forEach(container => {
            try {
                // Проверяем, не инициализирован ли уже
                if (!container.dataset.npkInitialized) {
                    new NPKConcentrationCalculator(container);
                    container.dataset.npkInitialized = 'true';
                }
            } catch (error) {
                console.error('Error initializing NPK calculator:', error);
                container.classList.add('npk-calculator-error');
            }
        });
    }
    
    // Инициализация при загрузке DOM
    document.addEventListener('DOMContentLoaded', initCalculators);
    
    // Экспорт для ручной инициализации
    window.npkInitCalculators = initCalculators;
    
})();
