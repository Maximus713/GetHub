```php
define('NPK_PLUGIN_URL', plugin_dir_url(__FILE__));

define('NPK_JS_URL', NPK_CONCENTRATION_MODULE_URL . 'assets/js/calculator.js');

 * Инициализация плагина NPK Calculators
 */

    // 1. Загружаем БАЗУ ДАННЫХ (главную или заглушку)
    if (file_exists(NPK_DATABASE_PATH)) {
        require_once NPK_DATABASE_PATH;
    } elseif (file_exists(NPK_DATABASE_STUB_PATH)) {
        require_once NPK_DATABASE_STUB_PATH;
    }
    
    // 2. Загружаем модуль концентрации
    if (file_exists(NPK_MODULE_PATH)) {
        require_once NPK_MODULE_PATH;
        
        // 3. Инициализируем модуль
        if (class_exists('NPK_Concentration_Module')) {
            NPK_Concentration_Module::get_instance();
        }
    }
}

// Ранняя инициализация на plugins_loaded
add_action('plugins_loaded', 'npk_calculators_init', 5);
```

#### **3.7. ФАЙЛ: npk-calculators/module-concentration/concentration-module.php**
**РОЛЬ:** WordPress интеграция модуля концентрации
**ЗАДАЧА:** Создать синглтон-модуль, который регистрирует все WordPress хуки.

```php
<?php
/**
 * Модуль концентрации для NPK Calculators
 * WordPress синглтон-модуль
 * На этапе ТЗ-1 регистрирует хуки, но логика не реализована
 */

if (!defined('ABSPATH')) {
    exit;
}

final class NPK_Concentration_Module {
    
    private static $instance = null;
    
    /**
     * Приватный конструктор
     */
    private function __construct() {
        $this->load_dependencies();
        $this->init_hooks();
    }
    
    /**
     * Получить экземпляр класса (синглтон)
     * @return self
     */
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    /**
     * Загрузить зависимости модуля
     */
    private function load_dependencies() {
        // Все классы должны быть в includes/
        $classes = [
            'class-calculation-engine.php',
            'class-data-adapter.php',
            'class-ajax-handler.php',
            'class-shortcode.php'
        ];
        
        foreach ($classes as $class_file) {
            $file_path = NPK_INCLUDES_PATH . $class_file;
            if (file_exists($file_path)) {
                require_once $file_path;
            }
        }
    }
    
    /**
     * Инициализировать хуки WordPress
     */
    private function init_hooks() {
        // Регистрация скриптов и стилей
        add_action('wp_enqueue_scripts', [$this, 'register_assets']);
        
        // Инициализация модуля на init
        add_action('init', [$this, 'init_module']);
    }
    
    /**
     * Регистрация скриптов и стилей модуля
     */
    public function register_assets() {
        // JavaScript калькулятора
        wp_register_script(
            'npk-concentration-calculator-js',
            NPK_JS_URL,
            [],
            '1.0.0',
            true
        );
        
        // CSS стили калькулятора
        wp_register_style(
            'npk-concentration-calculator-css',
            NPK_CSS_URL,
            [],
            '1.0.0'
        );
    }
    
    /**
     * Инициализация модуля (вызывается на init)
     */
    public function init_module() {
        // Регистрируем шорткод
        add_shortcode('npk_concentration_calculator', [NPK_Concentration_Shortcode::get_instance(), 'render']);
        
        // Регистрируем AJAX endpoint
        add_action('wp_ajax_npk_concentration_calculate', ['NPK_Concentration_AJAX_Handler', 'handle_request']);
        add_action('wp_ajax_nopriv_npk_concentration_calculate', ['NPK_Concentration_AJAX_Handler', 'handle_request']);
    }
    
    // Запрещаем клонирование и восстановление из сериализации
    private function __clone() {}
    public function __wakeup() {}
}
```

**СВЯЗИ:** Вызывается из npk-calculators.php, вызывает все 4 класса
**ПРОВЕРКА:** Модуль должен зарегистрировать шорткод и AJAX endpoint
