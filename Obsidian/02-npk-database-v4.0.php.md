    /**
     * Нормализует вектор NPK (сумма компонентов = 1)
     * Вспомогательный метод для расчёта коэффициента Отиаи
     * 
     * @param array $npk Вектор NPK [N, P, K]
     * @return array Нормализованный вектор
     */
    private static function normalize_npk($npk) {
        if (!is_array($npk) || count($npk) !== 3) {
            return [0, 0, 0];
        }
        
        $sum = array_sum($npk);
        
        // Избегаем деления на ноль
        if ($sum == 0) {
            return [0, 0, 0];
        }
        
        return [
            $npk[0] / $sum,
            $npk[1] / $sum,
            $npk[2] / $sum
        ];
    }
    
    /**
     * Возвращает отсортированный список культур для интерфейса выбора
     * Формат: ['tomato' => '🍅 Томаты', 'cucumber' => '🥒 Огурцы', ...]
     * 
     * @return array Ассоциативный массив [ID_культуры => "Наименование с эмодзи"]
     */
    public static function get_crops_list() {
        $crops = self::get_crops();
        $list = [];
        
        foreach ($crops as $crop_id => $crop_data) {
            $list[$crop_id] = $crop_data['emoji'] . ' ' . $crop_data['name'];
        }
        
        // Сортировка по алфавиту (по названию культуры)
        asort($list);
        
        return $list;
    }
    
    /**
     * Возвращает список фаз роста для конкретной культуры
     * Формат: ['seedling' => '🌱 Рассада', 'blooming' => '🌸 Цветение', ...]
     * 
     * @param string $crop_id Идентификатор культуры (например, 'tomato')
     * @return array Ассоциативный массив [ID_фазы => "Наименование с эмодзи"] или пустой массив
     */
    public static function get_phases_list($crop_id) {
        $crops = self::get_crops();
        
        if (!isset($crops[$crop_id]) || !isset($crops[$crop_id]['phases'])) {
            return [];
        }
        
        $phases = [];
        foreach ($crops[$crop_id]['phases'] as $phase_id => $phase_data) {
            $phases[$phase_id] = $phase_data['emoji'] . ' ' . $phase_data['name'];
        }
        
        return $phases;
    }
    
    /**
     * Возвращает эталонные потребности культуры на определённой фазе роста
     * 
     * @param string $crop_id Идентификатор культуры (например, 'tomato')
     * @param string $phase_id Идентификатор фазы (например, 'blooming')
     * @return array|false Массив с данными или false если не найдено
     */
    public static function get_crop_phase_requirements($crop_id, $phase_id) {
        $crops = self::get_crops();
        
        // Проверка существования культуры и фазы
        if (!isset($crops[$crop_id]['phases'][$phase_id])) {
            return false;
        }
        
        $phase_data = $crops[$crop_id]['phases'][$phase_id];
        $crop_data = $crops[$crop_id];
        
        return [
            // Основные данные
            'npk' => $phase_data['npk'],                         // [1.0, 2.0, 1.0]
            'npk_normalized' => self::normalize_npk($phase_data['npk']), // Нормализованный вектор
            'priority' => $phase_data['priority'] ?? [],         // ['P', 'Ca', 'микроэлементы']
            'notes' => $phase_data['notes'] ?? '',
            
            // Метаданные для интерфейса
            'crop_name' => $crop_data['name'],                   // "Томаты"
            'crop_emoji' => $crop_data['emoji'],                 // "🍅"
            'phase_name' => $phase_data['name'],                 // "Цветение и завязь"
            'phase_emoji' => $phase_data['emoji'],               // "🌸"
            
            // Идентификаторы
            'crop_id' => $crop_id,
            'phase_id' => $phase_id
        ];
    }
