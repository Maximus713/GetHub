```javascript
/**
 * ШАБЛОН: Отправка AJAX запроса
 * КОПИРУЙТЕ ЭТОТ КОД ДЛЯ ВСЕХ AJAX ЗАПРОСОВ
 * 
 * @param {Object} data - Данные для расчёта
 * @returns {Promise} Promise с ответом сервера
 */
async function sendCalculationRequest(data) {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 10000); // 10 сек timeout
    
    try {
        // ⚠️ ВАЖНО: ТОЛЬКО такой формат (ТЗ Приложение A)
        const response = await fetch(window.ajaxurl || window.NPK_Concentration_Ajax?.url, {
            method: 'POST',
            headers: { 
                'Content-Type': 'application/json' // ТОЛЬКО JSON!
            },
            body: JSON.stringify({ // ТОЛЬКО JSON.stringify!
                action: 'npk_concentration_calculate',
                nonce: window.NPK_Concentration_Ajax?.nonce || '',
                fertilizer_id: data.fertilizerId,
                concentration: data.concentration,
                water_volume: data.waterVolume,
                ground_type: data.groundType
            }),
            signal: controller.signal
        });
        
        clearTimeout(timeoutId);
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const result = await response.json();
        
        // Проверка структуры ответа
        if (!result || typeof result !== 'object') {
            throw new Error('Invalid response format');
        }
        
        return result;
        
    } catch (error) {
        clearTimeout(timeoutId);
        
        // ⚠️ Fallback по ТЗ п.7.4
        console.warn('AJAX error, using fallback:', error.message);
        
        return {
            success: false,
            data: {
                message: 'Используем локальные расчёты',
                is_fallback: true,
                local_data: data // Для локальных расчётов
            }
        };
    }
}
```

```php
<?php
/**
 * ШАБЛОН: AJAX_Handler для WordPress
 * КОПИРУЙТЕ ЭТУ СТРУКТУРУ
 */

class NPK_Concentration_AJAX_Handler {
    
    private static $instance = null;
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        add_action('wp_ajax_npk_concentration_calculate', [$this, 'handle_calculation']);
        add_action('wp_ajax_nopriv_npk_concentration_calculate', [$this, 'handle_calculation']);
    }
    
    /**
     * Обработчик AJAX запроса
     * ШАБЛОН: Всегда возвращает JSON
     */
    public function handle_calculation() {
        // 1. Проверка nonce (безопасность)
        if (!check_ajax_referer('npk_concentration_nonce', 'nonce', false)) {
            wp_send_json_error(['message' => 'Nonce verification failed'], 403);
            return;
        }
        
        // 2. Получение и валидация JSON
        $json_input = file_get_contents('php://input');
        $data = json_decode($json_input, true);
        
        if (json_last_error() !== JSON_ERROR_NONE) {
            wp_send_json_error(['message' => 'Invalid JSON format'], 400);
            return;
        }
        
        // 3. Проверка обязательных полей (ТЗ п.7.3.2)
        $required_fields = ['fertilizer_id', 'concentration', 'water_volume', 'ground_type'];
        foreach ($required_fields as $field) {
            if (!isset($data[$field])) {
                wp_send_json_error(['message' => "Missing required field: {$field}"], 400);
                return;
            }
        }
        
        // 4. Валидация типов и значений
        $fertilizer_id = absint($data['fertilizer_id']);
        $concentration = floatval($data['concentration']);
        $water_volume = floatval($data['water_volume']);
        $ground_type = sanitize_text_field($data['ground_type']);
        
        // 5. Дополнительная валидация
        if (!in_array($ground_type, ['dry', 'medium', 'wet'], true)) {
            wp_send_json_error(['message' => 'Invalid ground_type'], 400);
            return;
        }
        
        if ($concentration < 0.05 || $concentration > 1.5) {
            wp_send_json_error(['message' => 'Concentration out of range'], 400);
            return;
        }
        
        // 6. Вызов Calculation_Engine (единственный источник логики)
        try {
            $result = NPK_Concentration_Calculation_Engine::calculate(
                $fertilizer_id,
                $concentration,
                $water_volume,
                $ground_type
            );
            
            // 7. Успешный ответ (ТЗ п.7.3.3)
            wp_send_json_success($result);
            
        } catch (Exception $e) {
            // 8. Ошибка в логике (ТЗ п.7.3.4)
            wp_send_json_error([
                'message' => 'Calculation error',
                'debug' => $e->getMessage() // Только для разработки
            ], 500);
        }
        
        wp_die(); // Обязательно для WordPress AJAX
    }
}

// Инициализация
NPK_Concentration_AJAX_Handler::get_instance();
?>
```

```php
class NPK_Concentration_Calculation_Engine {

     * Определение state для концентрации
     * ТОЧНАЯ логика из ТЗ п.4.3
     * 
     * @param float $concentration - Концентрация в %
     * @param string $ground_type - Тип грунта
     * @return array - Массив состояния
     */
    private static function determine_state(float $concentration, string $ground_type): array {
        // 1. Получаем профиль из БД
        $profiles = NPK_Database::get_concentration_profiles();
        
        if (!isset($profiles[$ground_type])) {
            throw new InvalidArgumentException("Invalid ground type: {$ground_type}");
        }
        
        $profile = $profiles[$ground_type];
        $states = $profile['states'];
        $states_count = count($states);
        
        // 2. Проверка по всем состояниям (ТЗ п.4.3)
        for ($i = 0; $i < $states_count; $i++) {
            $state = $states[$i];
            $range = $state['range'];
            $is_last = ($i === $states_count - 1);
            
            if ($is_last) {
                // Последнее состояние: lower ≤ X ≤ upper
                if ($concentration >= $range[0] && $concentration <= $range[1]) {
                    return $state;
                }
            } else {
                // Обычное состояние: lower ≤ X < upper
                if ($concentration >= $range[0] && $concentration < $range[1]) {
                    return $state;
                }
            }
        }
        
        // 3. Fallback: возвращаем последнее состояние (ТЗ п.4.3, пункт 4)
```

```javascript
/**
 * ШАБЛОН: Определение предварительного state в UI
 * ТОЛЬКО ДЛЯ UX! Финальный state придёт с сервера
 * 
 * @param {number} concentration - Концентрация в %
 * @param {Object} profile - Профиль грунта из БД
 * @returns {Object} - Предварительное состояние
 */
function getPreviewState(concentration, profile) {
    if (!profile || !profile.states || !Array.isArray(profile.states)) {
        console.error('Invalid profile data');
        return null;
    }
    
    const states = profile.states;
    const statesCount = states.length;
    
    // ⚠️ ТОЧНО ТАКАЯ ЖЕ логика как в PHP (ТЗ п.4.3)
    for (let i = 0; i < statesCount; i++) {
        const state = states[i];
        const range = state.range;
        const isLast = (i === statesCount - 1);
        
        if (isLast) {
            // Последнее состояние: lower ≤ X ≤ upper
            if (concentration >= range[0] && concentration <= range[1]) {
                return {
                    ...state,
                    is_preview: true // Помечаем как предварительное
                };
            }
        } else {
            // Обычное состояние: lower ≤ X < upper
            if (concentration >= range[0] && concentration < range[1]) {
                return {
                    ...state,
                    is_preview: true // Помечаем как предварительное
                };
            }
        }
    }
    
    // Fallback: последнее состояние
    return {
        ...states[statesCount - 1],
        is_preview: true
    };
}
```

## **ШАБЛОН 3: ЦВЕТОВАЯ ИНТЕРПОЛЯЦИЯ (ТЗ п.6.4)**

```javascript
/**
 * ШАБЛОН: Интерполяция цвета поля концентрации
 * ТОЛЬКО ВИЗУАЛЬНЫЙ ЭФФЕКТ, не влияет на логику
 * 
 * @param {number} concentration - Текущая концентрация
 * @param {Object} state - Текущее состояние (предварительное или финальное)
 * @returns {string} - Цвет в формате RGB
 */
function interpolateConcentrationColor(concentration, state) {
    if (!state || !state.range || !state.color) {
        return '#ffffff'; // Белый по умолчанию
    }
    
    const [rangeStart, rangeEnd] = state.range;
    const stateColor = state.color;
    
    // 1. Преобразуем hex цвет в RGB
    const hexToRgb = (hex) => {
        const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
        return result ? {
            r: parseInt(result[1], 16),
            g: parseInt(result[2], 16),
            b: parseInt(result[3], 16)
        } : { r: 0, g: 0, b: 0 };
    };
    
    // 2. Цвет состояния
    const stateRgb = hexToRgb(stateColor);
    
    // 3. Начальный цвет (белый)
    const whiteRgb = { r: 255, g: 255, b: 255 };
    
    // 4. Рассчитываем вес (0 = начало диапазона, 1 = конец)
    let weight = 0;
    if (rangeEnd > rangeStart) {
        weight = (concentration - rangeStart) / (rangeEnd - rangeStart);
        weight = Math.max(0, Math.min(1, weight)); // Ограничиваем 0-1
    }
    
    // 5. Интерполяция (ТЗ п.6.4)
    const r = Math.round(whiteRgb.r + (stateRgb.r - whiteRgb.r) * weight);
    const g = Math.round(whiteRgb.g + (stateRgb.g - whiteRgb.g) * weight);
    const b = Math.round(whiteRgb.b + (stateRgb.b - whiteRgb.b) * weight);
    
    return `rgb(${r}, ${g}, ${b})`;
}

/**
 * ШАБЛОН: Синхронное обновление цветов всех элементов
 * ТЗ п.6.4.2: Все элементы меняют цвет ОДНОВРЕМЕННО
 */
function updateAllColorsSynchronously(newState, concentration) {
    // ⚠️ ВАЖНО: Используем requestAnimationFrame для синхронности
    requestAnimationFrame(() => {
        // 1. Слайдер (заливка слева от ползунка)
        const sliderTrack = document.querySelector('.npk-slider-track');
        if (sliderTrack) {
            sliderTrack.style.backgroundColor = newState.color;
        }
        
        // 2. Метка над слайдером
        const stateLabel = document.querySelector('.npk-state-label');
        if (stateLabel) {
            stateLabel.style.backgroundColor = newState.color;
            stateLabel.textContent = newState.name;
        }
        
        // 3. Поле ввода концентрации (интерполяция)
        const concentrationInput = document.querySelector('.npk-concentration-input');
        if (concentrationInput) {
            concentrationInput.style.backgroundColor = interpolateConcentrationColor(
                concentration, 
                newState
            );
        }
        
        // 4. Блок результатов
        const resultState = document.querySelector('.npk-result-state');
        if (resultState) {
            resultState.style.backgroundColor = newState.color;
            resultState.textContent = `${newState.emoji} ${newState.name}`;
        }
    });
}
```

## **ШАБЛОН 4: ЛОКАЛЬНЫЙ РАСЧЁТ МАССЫ (ТЗ п.6.2)**

```javascript
/**
 * ШАБЛОН: Локальный (предварительный) расчёт массы
 * ТОЛЬКО ДЛЯ UX! Финальный расчёт на сервере
 * 
 * @param {number} concentration - Концентрация в %
 * @param {number} waterVolume - Объём воды в литрах
 * @param {Object} fertilizer - Данные удобрения
 * @returns {number} - Предварительная масса в граммах
 */
function calculateLocalMass(concentration, waterVolume, fertilizer) {
    // ⚠️ УПРОЩЁННАЯ формула для мгновенного UX
    // Не содержит компенсацию растворимости!
    // Не является финальным результатом!
    
    if (!fertilizer || typeof fertilizer.solubility_total !== 'number') {
        // Fallback: простая формула без растворимости
        return Math.round((concentration * waterVolume * 1000) / 100);
    }
    
    // Упрощённый расчёт с растворимостью (для лучшего UX)
    const dissolvedMass = (concentration * waterVolume * 1000) / 100;
    const compensation = 100 / Math.max(1, fertilizer.solubility_total);
    const mass = dissolvedMass * compensation;
    
    return Math.round(mass);
}

/**
 * ШАБЛОН: Расчёт концентрации NPK (ТЗ п.6.3)
 * ТОЛЬКО ДЛЯ ОТОБРАЖЕНИЯ, не для логики
 */
function calculateNPKConcentration(totalConcentration, fertilizer) {
    if (!fertilizer || !fertilizer.npk_sum) {
        return 0;
    }
    
    // Формула: общая_концентрация × (сумма_NPK / 100)
    const npkPercentage = fertilizer.npk_sum / 100;
    const npkConcentration = totalConcentration * npkPercentage;
    
    // Округление до 3 знаков как в ТЗ п.5
    return Math.round(npkConcentration * 1000) / 1000;
}
```

## **ШАБЛОН 5: DEBOUNCE И УПРАВЛЕНИЕ ЗАПРОСАМИ**

```javascript
/**
 * ШАБЛОН: Debounce управление запросами
 * ТЗ п.6.2: разные стратегии для разных событий
 */
class RequestManager {
    constructor() {
        this.currentController = null;
        this.debounceTimers = {};
        this.lastRequestData = null;
    }
    
    /**
     * Отправка запроса с debounce
     * @param {Object} data - Данные для запроса
     * @param {string} type - Тип события ('input', 'ground_change', 'slider_drag')
     * @param {Function} callback - Колбэк при успехе
     */
    sendRequest(data, type, callback) {
        // 1. Отменяем предыдущий запрос
        if (this.currentController) {
            this.currentController.abort();
        }
        
        // 2. Создаём новый AbortController
        this.currentController = new AbortController();
        
        // 3. Стратегия debounce по типу (ТЗ п.6.2)
        let debounceTime = 500; // По умолчанию
        
        switch (type) {
            case 'input':
            case 'concentration_change':
                debounceTime = 500; // 500мс после последнего изменения
                break;
                
            case 'ground_change':
            case 'fertilizer_change':
                debounceTime = 0; // Немедленно
                break;
                
            case 'slider_drag':
                // Запрос только после отпускания
                this.lastRequestData = { data, callback };
                return;
                
            case 'slider_release':
                // Используем сохранённые данные
                if (this.lastRequestData) {
                    data = this.lastRequestData.data;
                    callback = this.lastRequestData.callback;
                    this.lastRequestData = null;
                }
                debounceTime = 0;
                break;
        }
        
        // 4. Debounce
        clearTimeout(this.debounceTimers[type]);
        
        this.debounceTimers[type] = setTimeout(() => {
            this.executeRequest(data, callback);
        }, debounceTime);
    }
    
    /**
     * Выполнение запроса
     */
    async executeRequest(data, callback) {
        try {
            const result = await sendCalculationRequest(data);
            
            if (result.success && callback) {
                callback(result.data);
            } else if (!result.success && result.data?.is_fallback) {
                // Fallback: используем локальные расчёты
                this.handleFallback(data, callback);
            }
            
        } catch (error) {
            console.warn('Request failed:', error.message);
            this.handleFallback(data, callback);
        }
    }
    
    /**
     * Fallback на локальные расчёты (ТЗ п.7.4)
     */
    handleFallback(data, callback) {
        // 1. Показываем уведомление
        this.showFallbackNotification();
        
        // 2. Используем локальные расчёты
        const localResult = this.calculateLocally(data);
        
        // 3. Вызываем колбэк с локальными данными
        if (callback) {
            callback({
                ...localResult,
                is_fallback: true,
                warning: 'Используем локальные расчёты'
            });
        }
        
        // 4. Повторная попытка через 5 секунд
        setTimeout(() => {
            this.sendRequest(data, 'retry', callback);
        }, 5000);
    }
    
    /**
     * Локальный расчёт при отключённом сервере
     */
    calculateLocally(data) {
        // Используем шаблон 4 для локальных расчётов
        const fertilizer = this.getFertilizerById(data.fertilizerId);
        const profile = this.getProfileByType(data.groundType);
        
        const previewState = getPreviewState(data.concentration, profile);
        const localMass = calculateLocalMass(
            data.concentration, 
            data.waterVolume, 
            fertilizer
        );
        
        return {
            fertilizer_mass: localMass,
            concentration: data.concentration,
            water_volume: data.waterVolume,
            state: previewState,
            is_local: true
        };
    }
}
```

```javascript
/**
 * ШАБЛОН: Безопасное обновление UI
 * ТЗ Приложение A: Все изменения DOM внутри requestAnimationFrame
 */

class UIUpdater {
    constructor() {
        this.pendingUpdates = new Map();
        this.isUpdating = false;
    }
    
    /**
     * Постановка обновления в очередь
     * @param {string} elementId - ID элемента
     * @param {Function} updateFn - Функция обновления
     */
    queueUpdate(elementId, updateFn) {
        this.pendingUpdates.set(elementId, updateFn);
        this.scheduleUpdate();
    }
    
    /**
     * Планирование обновления
     */
    scheduleUpdate() {
        if (!this.isUpdating) {
            this.isUpdating = true;
            requestAnimationFrame(() => this.processUpdates());
        }
    }
    
    /**
     * Обработка всех обновлений в одном кадре
     */
    processUpdates() {
        // ⚠️ ВСЕ изменения DOM здесь
        this.pendingUpdates.forEach((updateFn, elementId) => {
            try {
                updateFn();
            } catch (error) {
                console.error(`Error updating ${elementId}:`, error);
            }
        });
        
        this.pendingUpdates.clear();
        this.isUpdating = false;
    }
    
    /**
     * ШАБЛОН: Обновление нескольких элементов синхронно
     */
    updateMultiple(elements) {
        requestAnimationFrame(() => {
            // 1. Сначала скрываем изменения
            elements.forEach(el => {
                if (el.element) {
                    el.element.style.visibility = 'hidden';
                }
            });
            
            // 2. Применяем все изменения
            elements.forEach(el => {
                if (el.element && el.updates) {
                    Object.entries(el.updates).forEach(([property, value]) => {
                        el.element.style[property] = value;
                    });
                }
            });
            
            // 3. Показываем всё одновременно
            elements.forEach(el => {
                if (el.element) {
                    el.element.style.visibility = 'visible';
                }
            });
        });
    }
}
```
