### **4.2. `includes/class-ajax-handler.php` (с интегрированным кешированием)**
```php
<?php
/**
 * Обработчик AJAX с интегрированным кешированием
 * ИСПРАВЛЕНИЕ 2: Кеширование в этом же классе
 */

if (!defined('ABSPATH')) {
    exit;
}

class FS_AJAX_Handler {
    
    /**
     * Инициализация AJAX хуков
     */
    public static function init() {
        add_action('wp_ajax_npk_fs_calculate', [__CLASS__, 'handle_calculation']);
        add_action('wp_ajax_nopriv_npk_fs_calculate', [__CLASS__, 'handle_calculation']);
    }
    
    /**
     * Обработчик расчёта с кешированием
     */
    public static function handle_calculation() {
        // Проверка nonce
        check_ajax_referer('npk_fs_calculate', 'nonce');
        
        $crop_id = sanitize_text_field($_POST['crop_id'] ?? '');
        $phase_id = sanitize_text_field($_POST['phase_id'] ?? '');
        
        // Генерация ключа кеша
        $cache_key = 'npk_fs_' . md5($crop_id . '_' . $phase_id);
        
        // Попытка получить из кеша
        $cached = self::get_cache($cache_key);
        if ($cached !== false) {
            wp_send_json_success($cached);
            return;
        }
        
        // Расчёт если нет в кеше
        $result = FS_Selection_Engine::find_matching_fertilizers($crop_id, $phase_id);
        
        // Сохранение в кеш (1 час)
        self::set_cache($cache_key, $result, HOUR_IN_SECONDS);
        
        wp_send_json_success($result);
    }
    
    /**
     * Получить данные из кеша
     * ИСПРАВЛЕНИЕ 2: Метод кеширования в этом классе
     */
    private static function get_cache($key) {
        return get_transient($key);
    }
    
    /**
     * Сохранить данные в кеш
     * ИСПРАВЛЕНИЕ 2: Метод кеширования в этом классе
     */
    private static function set_cache($key, $data, $expiration) {
        set_transient($key, $data, $expiration);
    }
}
```

### **4.3. `assets/js/selector.js` (с исправленными префиксами)**
```js
/**
 * JavaScript логика модуля подбора удобрений
 * ИСПРАВЛЕНИЕ 1: Уникальные префиксы FS_
 */

(function() {
    'use strict';
    
    // ИСПРАВЛЕНИЕ 1: FS_ вместо NPK_FS_
    window.FS = window.FS || {};
    
    // Основные данные (ИСПРАВЛЕНИЕ 1)
    FS.Data = window.FS_Data || {};
    
    // AJAX утилиты (ИСПРАВЛЕНИЕ 1)
    FS.Ajax = {
        calculate: function(cropId, phaseId) {
            const formData = new FormData();
            formData.append('action', 'npk_fs_calculate');
            formData.append('nonce', FS.Data.nonce || '');
            formData.append('crop_id', cropId);
            formData.append('phase_id', phaseId);
            
            return fetch(FS.Data.ajax_url || window.ajaxurl, {
                method: 'POST',
                body: formData
            });
        }
    };
    
    // Инициализация
    FS.init = function() {
        console.log('Fertilizer Selector (FS) initialized');
    };
    
    // Автоматическая инициализация
    document.addEventListener('DOMContentLoaded', FS.init);
    
})();
```

### **4.4. `includes/class-data-adapter.php` (с исправленными префиксами)**
```php
<?php
/**
 * Адаптер данных для JavaScript
 * ИСПРАВЛЕНИЕ 1: Правильные префиксы
 */

if (!defined('ABSPATH')) {
    exit;
}

class FS_Data_Adapter {
    
    /**
     * Получить данные для инициализации JavaScript
     * ИСПРАВЛЕНИЕ 1: FS_Data а не NPK_FS_Data
     */
    public static function get_js_data() {
        return [
            'ajax_url' => admin_url('admin-ajax.php'),
            'nonce' => wp_create_nonce('npk_fs_calculate'),
            // ИСПРАВЛЕНИЕ 1: Ключ FS_Data
            'crops' => self::get_crops_data(),
            'fertilizers' => self::get_fertilizers_data(),
        ];
    }
    
    /**
     * Подготовить данные культур для JS
     */
    private static function get_crops_data() {
        $crops = NPK_Database::get_crops_list();
        $data = [];
        
        foreach ($crops as $crop_id => $crop_name) {
            $data[$crop_id] = [
                'id' => $crop_id,
                'name' => $crop_name,
                'phases' => NPK_Database::get_phases_list($crop_id),
            ];
        }
        
        return $data;
    }
    
    /**
     * Подготовить данные удобрений для JS
     */
    private static function get_fertilizers_data() {
        $fertilizers = NPK_Database::get_fertilizers();
        $data = [];
        
        foreach ($fertilizers as $fert) {
            $data[] = [
                'id' => $fert['id'],
                'name' => $fert['name'],
                'npk' => $fert['npk'],
                'type' => $fert['type'],
                'additives' => $fert['additives'] ?? [],
            ];
        }
        
        return $data;
    }
}
```
