```php
class Имя_Модуля {
    private static $instance = null;
    private $version = '1.0.0';
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        // ⚡ РАННЯЯ регистрация скриптов (приоритет 1)
        add_action('wp_enqueue_scripts', [$this, 'register_assets'], 1);
        
        // Загрузка зависимостей
        $this->load_dependencies();
        
        // Инициализация на init
        add_action('init', [$this, 'init']);
    }
    
    private function load_dependencies() {
        // Подключаем все необходимые классы
    }
    
    public function register_assets() {
        // ✅ ВСЕГДА регистрируем, даже если шорткод не используется
        wp_register_style(
            'npk-префикс-css',
            plugin_dir_url(__FILE__) . 'assets/css/файл.css',
            [],
            $this->version,
            'all'
        );
        
        wp_register_script(
            'npk-префикс-js',
            plugin_dir_url(__FILE__) . 'assets/js/файл.js',
            [],
            $this->version,
            true
        );
        
        // ✅ Передача данных в JS (если данные готовы)
        if ($данные = $this->get_js_data()) {
            wp_localize_script('npk-префикс-js', 'Объект_Данных', $данные);
        }
    }
    
    public function init() {
        // Регистрация шорткода
        add_shortcode('имя_шорткода', [$this, 'render_shortcode']);
    }
    
    public function render_shortcode($atts = [], $content = null) {
        // ✅ ВСЕГДА enqueue, БЕЗ проверок
        wp_enqueue_style('npk-префикс-css');
        wp_enqueue_script('npk-префикс-js');
        
        // Если данные не были переданы в register_assets, передаём здесь
        if (!wp_script_is('npk-префикс-js', 'done')) {
            $данные = $this->get_js_data();
            wp_add_inline_script('npk-префикс-js', 'window.Объект_Данных = ' . json_encode($данные) . ';', 'before');
        }
        
        // Генерация HTML
        ob_start();
        include __DIR__ . '/templates/шаблон.php';
        return ob_get_clean();
    }
}
```

## ШАБЛОН 4: AJAX ОБРАБОТЧИК

```php
class Имя_AJAX_Обработчика {
    private static $instance = null;
    private $ajax_action = 'имя_действия';
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        // Для авторизованных пользователей
        add_action("wp_ajax_{$this->ajax_action}", [$this, 'handle_ajax']);
        
        // Для неавторизованных пользователей
        add_action("wp_ajax_nopriv_{$this->ajax_action}", [$this, 'handle_ajax']);
    }
    
    public function handle_ajax() {
        // 1. Проверка nonce
        check_ajax_referer('nonce_действие', 'nonce');
        
        // 2. Валидация данных
        $данные = $this->validate_input($_POST);
        
        // 3. Обработка
        try {
            $результат = $this->process($данные);
            wp_send_json_success($результат);
        } catch (Exception $e) {
            wp_send_json_error(['message' => $e->getMessage()]);
        }
    }
    
    private function validate_input($input) {
        // Валидация всех полей
        return [
            'поле' => sanitize_text_field($input['поле']),
            'число' => floatval($input['число'])
        ];
    }
}
```

```javascript
if (typeof window.Объект_Данных === 'undefined') {

    return;

// 2. AJAX запрос (ТОЛЬКО JSON)
async function sendRequest(data) {
    try {
        const response = await fetch(window.ajaxurl, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                action: 'имя_действия',
                nonce: window.Объект_Данных.ajax.nonce,
                ...data
            })
        });
        
        return await response.json();
    } catch (error) {
        console.error('AJAX ошибка:', error);
        throw error;
    }
}

// 3. Обновление UI (ТОЛЬКО через requestAnimationFrame)
function updateUI(element, newContent, newColor) {
    requestAnimationFrame(() => {
        element.textContent = newContent;
        element.style.backgroundColor = newColor;
    });
}

// 4. Дебаунс для обработки ввода
function createDebounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}
```

## КРИТИЧЕСКИЕ ПРАВИЛА

### ❌ НИКОГДА НЕ ДЕЛАТЬ:
1. `wp_register_script()` внутри шорткода
2. Проверку `wp_script_is()` в `render()` методе
3. `new FormData()` для AJAX (только `JSON.stringify()`)
4. Прямое обновление DOM без `requestAnimationFrame()`
5. Использование `wp_enqueue_scripts` без приоритета 1 для регистрации

### ✅ ВСЕГДА ДЕЛАТЬ:
1. Регистрировать скрипты в конструкторе класса
2. Enqueue скрипты в шорткоде БЕЗ условий
3. Использовать `application/json` для AJAX
4. Добавлять `wp_nonce_field()` или `wp_create_nonce()`
5. Проверять существование классов перед их использованием

---

## ЧАСТЫЕ ОШИБКИ И ИСПРАВЛЕНИЯ

### Ошибка: Шорткод показывает код вместо контента
**Причина:** Скрипты не зарегистрированы до вызова шорткода
**Решение:** Регистрация в конструкторе с `add_action('wp_enqueue_scripts', ..., 1)`

### Ошибка: Страница перезагружается при AJAX
**Причина:** Использование `FormData` или `x-www-form-urlencoded`
**Решение:** Только `JSON.stringify()` и `application/json`

### Ошибка: Моргание интерфейса
**Причина:** Прямое обновление DOM
**Решение:** Использовать `requestAnimationFrame()`

### Ошибка: Скрипты не загружаются
**Причина:** Условная логика в шорткоде
**Решение:** Всегда вызывать `wp_enqueue_script()` без проверок
