### **УРОВЕНЬ 3: MUTATIONOBSERVER ДЛЯ INLINE-STYLES**
```javascript
// Защищаем ТОЛЬКО элементы калькулятора
const calcContainer = document.querySelector('.npk-concentration-calculator');
if (calcContainer) {
    const observer = new MutationObserver((mutations) => {
        mutations.forEach((mutation) => {
            // Удаляем ТОЛЬКО inline-стили на элементах калькулятора
            if (mutation.type === 'attributes' && 
                mutation.attributeName === 'style' &&
                mutation.target.closest('.npk-concentration-calculator')) {
                
                // Проверяем, не наш ли это стиль?
                const style = mutation.target.getAttribute('style');
                if (style && style.includes('elementor')) {
                    mutation.target.removeAttribute('style');
                }
            }
        });
    });
    
    observer.observe(calcContainer, {
        attributes: true,
        subtree: true,
        attributeFilter: ['style']
    });
}
```

```php
add_action('wp_enqueue_scripts', 'npk_protect_from_elementor', 999999);

    // Удаляем ТОЛЬКО основные стили Elementor
    wp_dequeue_style('elementor-frontend');

    // - elementor-icons
    // - elementor-animations  
    // - elementor-font-awesome
}

// 2. ДОБАВЛЯЕМ ЗАЩИТНЫЕ CSS ПРАВИЛА
add_action('wp_head', 'npk_add_protection_css', 100);
function npk_add_protection_css() {
    ?>

            background: white !important;

        .npk-concentration-calculator .npk-fertilizer-select {
            border: 1px solid #e0d8c9 !important;
            border-radius: 8px !important;
            padding: 12px 15px !important;
        }

    <?php
}

// 3. ВЫВОДИМ СТРАНИЦУ
get_header('minimal'); ?>

<!-- КОНТЕЙНЕР ИЗОЛЯЦИИ -->
<div class="npk-page-wrapper" data-npk-isolated="true">

        echo do_shortcode('[npk_calculator]');

    <footer class="npk-page-footer">
        <p>&copy; <?php echo date('Y'); ?> <?php bloginfo('name'); ?></p>
    </footer>
    
</div>

<?php 
// 4. ДОБАВЛЯЕМ ЗАЩИТНЫЙ JAVASCRIPT
add_action('wp_footer', 'npk_add_protection_js', 999);
function npk_add_protection_js() {
    ?>
    <script>
    (function() {
        'use strict';
        
        // Защищаем калькулятор от inline-стилей Elementor
        const calcElement = document.querySelector('.npk-concentration-calculator');
        if (!calcElement) return;
        
        // MutationObserver для удаления вредных inline-стилей
        const observer = new MutationObserver(function(mutations) {
            mutations.forEach(function(mutation) {
                if (mutation.type === 'attributes' && 
                    mutation.attributeName === 'style' &&
                    mutation.target.closest('.npk-concentration-calculator')) {
                    
                    const style = mutation.target.getAttribute('style');
                    // Удаляем ТОЛЬКО стили с elementor или border:none
                    if (style && (style.includes('elementor') || 
                                  style.includes('border: none') ||
                                  style.includes('border:none'))) {
                        mutation.target.removeAttribute('style');
                    }
                }
            });
        });
        
        // Запускаем наблюдение
        observer.observe(calcElement, {
            attributes: true,
            subtree: true,
            attributeFilter: ['style']
        });
        
        // Очистка через 10 секунд (больше не нужно)
        setTimeout(() => observer.disconnect(), 10000);
        
    })();
    </script>
    <?php
}

get_footer('minimal');
```
