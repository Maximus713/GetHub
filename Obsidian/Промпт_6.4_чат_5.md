// ============================================
// ДЕБАГ-ПРОВЕРКИ: class-selection-engine.php
// ============================================
// 1. Все пункты задания выполнены: ✅
//    - Метод find_matching_fertilizers() создан: ✅
//    - Получение эталона через NPK_Database: ✅
//    - Получение удобрений через NPK_Database: ✅
//    - Преобразование процентов (÷100): ✅
//    - Нормализация через NPK_Database::normalize_npk(): ✅
//    - Расчёт перекрытия (сумма min): ✅
//    - Порог 75% из констант: ✅
//    - Fallback-режим (3 лучших): ✅
//    - Подсказки (array_intersect): ✅
//    - Сортировка по % → по имени: ✅
//
// 2. Запреты соблюдены: ✅
//    - Нет прямых SQL-запросов: ✅
//    - Нет вызовов Data_Adapter/Shortcode/AJAX: ✅
//    - Нет собственной normalize_npk(): ✅
//
// 3. Чек-лист самопроверки: ✅ (12/12 пунктов)
// ============================================
<?php
/**
 * Selection Engine for Fertilizer Selector Module
 *
 * @package NPK_Calculators
 * @subpackage Fertilizer_Selector
 */
class NPK_Fertilizer_Selection_Engine {
    
    /**
     * Find matching fertilizers for crop and phase
     *
     * @param string $crop_id  Crop identifier
     * @param string $phase_id Phase identifier
     * @return array Sorted fertilizers with match percentages and tips
     */
    public static function find_matching_fertilizers($crop_id, $phase_id): array {
        // Get requirements
        $requirements = NPK_Database::get_crop_phase_requirements($crop_id, $phase_id);
        if (!$requirements) {
            return [];
        }
        
        $target_npk = $requirements['npk'];
        $priority = $requirements['priority'] ?? [];
        
        // Get all fertilizers
        $fertilizers = NPK_Database::get_fertilizers();
        
        // Get threshold
        $constants = NPK_Database::get_constants();
        $threshold = $constants['thresholds']['min_match_percentage'];
        
        $results = [];
        
        foreach ($fertilizers as $fertilizer) {
            // Convert percentages to relative values
            $fert_relative = array_map(function($p) {
                return $p / 100;
            }, $fertilizer['npk']);
            
            // Normalize vectors
            $norm_target = NPK_Database::normalize_npk($target_npk);
            $norm_fert = NPK_Database::normalize_npk($fert_relative);
            
            // Calculate overlap (Otiai algorithm)
            $overlap = 0;
            for ($i = 0; $i < 3; $i++) {
                $overlap += min($norm_target[$i], $norm_fert[$i]);
            }
            
            $percentage = round($overlap * 100, 1);
            
            // Generate tips
            $tips = [];
            if (!empty($priority) && !empty($fertilizer['additives'])) {
                $intersection = array_intersect($priority, $fertilizer['additives']);
                foreach ($intersection as $element) {
                    $tips[] = "Содержит {$element}, важный для {$requirements['phase_name']}";
                }
            }
            
            $results[] = [
                'id' => $fertilizer['id'],
                'name' => $fertilizer['name'],
                'npk' => $fertilizer['npk'],
                'type' => $fertilizer['type'],
                'match_percentage' => $percentage,
                'tips' => $tips,
                'fallback' => false
            ];
        }
        
        // Filter by threshold
        $passing = array_filter($results, function($item) use ($threshold) {
            return $item['match_percentage'] >= $threshold;
        });
        
        // Fallback mode
        if (count($passing) === 0) {
            usort($results, function($a, $b) {
                if ($a['match_percentage'] === $b['match_percentage']) {
                    return strcmp($a['name'], $b['name']);
                }
                return $b['match_percentage'] <=> $a['match_percentage'];
            });
            
            $fallback = array_slice($results, 0, 3);
            foreach ($fallback as &$item) {
                $item['fallback'] = true;
            }
            return $fallback;
        }
        
        // Sort passing results
        usort($passing, function($a, $b) {
            if ($a['match_percentage'] === $b['match_percentage']) {
                return strcmp($a['name'], $b['name']);
            }
            return $b['match_percentage'] <=> $a['match_percentage'];
        });
        
        return array_values($passing);
    }
}
