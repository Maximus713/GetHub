#### **ФАЙЛ: module-concentration/includes/class-ajax-handler.php**
**СУЩЕСТВУЮЩИЙ КОД:** Возвращает тестовый массив с `'test' => 'AJAX работает!'`
**РОЛЬ:** Транспорт данных между JavaScript и Calculation Engine
**ЗАДАЧА:** Заменить тестовую реализацию на реальный вызов Calculation Engine с валидацией входных данных

**КОД (РЕАЛИЗАЦИЯ):**
```php
<?php
/**
 * AJAX Handler для калькулятора концентрации
 * РЕАЛЬНАЯ РЕАЛИЗАЦИЯ с вызовом Calculation Engine
 */

if (!defined('ABSPATH')) {
    exit;
}

final class NPK_Concentration_AJAX_Handler {
    
    /**
     * Обработчик AJAX запросов - РЕАЛЬНАЯ РЕАЛИЗАЦИЯ
     * @throws Exception При ошибках валидации или расчёта
     */
    public static function handle_request(): void {
        // Логирование для отладки
        self::log_request();
        
        try {
            // 1. Валидация обязательных полей
            $data = self::validate_and_sanitize_input();
            
            // 2. Проверка существования Calculation Engine
            self::check_dependencies();
            
            // 3. Проверка типа грунта
            self::validate_ground_type($data['ground_type']);
            
            // 4. КРИТИЧЕСКАЯ ПРОВЕРКА: water_volume_max = 200
            self::check_water_volume_max();
            
            // 5. Вызов реального расчёта через Calculation Engine
            $result = NPK_Concentration_Calculation_Engine::calculate(
                $data['fertilizer_id'],
                $data['concentration'],
                $data['water_volume'],
                $data['ground_type']
            );
            
            // 6. КРИТИЧЕСКАЯ ПРОВЕРКА: 0.30% для влажного грунта = optimal
            self::check_initial_state($data['concentration'], $data['ground_type'], $result['state']['id']);
            
            // 7. Логирование результата
            self::log_result($result);
            
            // 8. Возврат реальных данных
            wp_send_json_success($result);
            
        } catch (InvalidArgumentException $e) {
            // Ошибки валидации (клиентские ошибки)
            wp_send_json_error(['message' => $e->getMessage()], 400);
            
        } catch (Exception $e) {
            // Серверные ошибки
            error_log('NPK AJAX Server Error: ' . $e->getMessage());
            wp_send_json_error([
                'message' => 'Внутренняя ошибка сервера при расчёте',
                'debug' => defined('WP_DEBUG') && WP_DEBUG ? $e->getMessage() : null
            ], 500);
        }
    }
    
    /**
     * Валидация и санация входных данных
     * @return array Валидированные данные
     * @throws InvalidArgumentException При невалидных данных
     */
    private static function validate_and_sanitize_input(): array {
        $required_fields = ['fertilizer_id', 'concentration', 'water_volume', 'ground_type'];
        $data = [];
        
        foreach ($required_fields as $field) {
            if (!isset($_POST[$field]) || $_POST[$field] === '') {
                throw new InvalidArgumentException("Отсутствует обязательное поле: {$field}");
            }
        }
        
        // Валидация fertilizer_id (целое положительное число)
        $fertilizer_id = (int) $_POST['fertilizer_id'];
        if ($fertilizer_id <= 0) {
            throw new InvalidArgumentException("Неверный ID удобрения: {$fertilizer_id}");
        }
        
        // Валидация concentration (число 0.05-1.5)
        $concentration = (float) $_POST['concentration'];
        if ($concentration < 0.05 || $concentration > 1.5) {
            throw new InvalidArgumentException("Концентрация должна быть в диапазоне 0.05-1.5%");
        }
        
        // Валидация water_volume (число > 0)
        $water_volume = (float) $_POST['water_volume'];
        if ($water_volume <= 0) {
            throw new InvalidArgumentException("Объём воды должен быть больше 0");
        }
        
        // Санация ground_type
        $ground_type = sanitize_text_field($_POST['ground_type']);
        
        return [
            'fertilizer_id' => $fertilizer_id,
            'concentration' => $concentration,
            'water_volume' => $water_volume,
            'ground_type' => $ground_type
        ];
    }
    
    /**
     * Проверка зависимостей (существование классов)
     * @throws Exception Если зависимости не загружены
     */
    private static function check_dependencies(): void {
        if (!class_exists('NPK_Concentration_Calculation_Engine')) {
            throw new Exception('Класс Calculation Engine не найден');
        }
        
        if (!class_exists('NPK_Database')) {
            throw new Exception('Класс базы данных не найден');
        }
        
        if (!method_exists('NPK_Concentration_Calculation_Engine', 'calculate')) {
            throw new Exception('Метод calculate() не найден в Calculation Engine');
        }
    }
    
    /**
     * Валидация типа грунта
     * @param string $ground_type Тип грунта
     * @throws InvalidArgumentException При невалидном типе грунта
     */
    private static function validate_ground_type(string $ground_type): void {
        $valid_types = ['dry', 'medium', 'wet'];
        
        if (!in_array($ground_type, $valid_types, true)) {
            throw new InvalidArgumentException(
                "Неверный тип грунта: '{$ground_type}'. Допустимые значения: " . 
                implode(', ', $valid_types)
            );
        }
    }
    
    /**
     * КРИТИЧЕСКАЯ ПРОВЕРКА: water_volume_max = 200
     * @throws Exception Если значение не соответствует ТЗ
     */
    private static function check_water_volume_max(): void {
        if (!class_exists('NPK_Database')) {
            return; // Не можем проверить
        }
        
        $constants = NPK_Database::get_constants();
        $water_volume_max = $constants['thresholds']['water_volume_max'] ?? null;
        
        if ($water_volume_max !== 200) {
            error_log("NPK WARNING: water_volume_max = {$water_volume_max}, should be 200 according to ТЗ");
            // Не выбрасываем исключение, только логируем
        }
    }
    
    /**
     * КРИТИЧЕСКАЯ ПРОВЕРКА: 0.30% для влажного грунта = optimal
     * @param float $concentration Концентрация
     * @param string $ground_type Тип грунта
     * @param string $state_id ID состояния
     */
    private static function check_initial_state(
        float $concentration, 
        string $ground_type, 
        string $state_id
    ): void {
        if (abs($concentration - 0.30) < 0.001 && $ground_type === 'wet' && $state_id !== 'optimal') {
            error_log("NPK WARNING: 0.30% for wet ground should be 'optimal', got: '{$state_id}'");
            // Логируем но не прерываем работу
        }
    }
    
    /**
     * Логирование входящего запроса
     */
    private static function log_request(): void {
        if (!defined('WP_DEBUG') || !WP_DEBUG) {
            return;
        }
        
        $log_data = [
            'timestamp' => current_time('mysql'),
            'fertilizer_id' => $_POST['fertilizer_id'] ?? 'not_set',
            'concentration' => $_POST['concentration'] ?? 'not_set',
            'water_volume' => $_POST['water_volume'] ?? 'not_set',
            'ground_type' => $_POST['ground_type'] ?? 'not_set'
        ];
        
        error_log('NPK AJAX Request: ' . json_encode($log_data));
    }
    
    /**
     * Логирование результата расчёта
     * @param array $result Результат расчёта
     */
    private static function log_result(array $result): void {
        if (!defined('WP_DEBUG') || !WP_DEBUG) {
            return;
        }
        
        $log_data = [
            'fertilizer_mass' => $result['fertilizer_mass'] ?? 0,
            'concentration' => $result['concentration'] ?? 0,
            'state_id' => $result['state']['id'] ?? 'unknown',
            'state_name' => $result['state']['name'] ?? 'unknown'
        ];
        
        error_log('NPK AJAX Result: ' . json_encode($log_data));
    }
    
    /**
     * Регистрация AJAX обработчиков (уже делается в модуле)
     */
    public static function register_ajax_handlers(): void {
        add_action('wp_ajax_npk_concentration_calculate', [__CLASS__, 'handle_request']);
        add_action('wp_ajax_nopriv_npk_concentration_calculate', [__CLASS__, 'handle_request']);
        
        if (defined('WP_DEBUG') && WP_DEBUG) {
            error_log('NPK: AJAX handlers registered: npk_concentration_calculate');
        }
    }
}
```
