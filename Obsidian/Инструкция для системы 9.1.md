// ============================================
// ДЕБАГ: module-fertilizer-selector/includes/class-cache-manager.php, задание №1
//
// Проверка перед выводом:
// [✅] все требования выполнены

<?php
/**
 * Класс для управления кешем модуля подбора удобрений
 */

if (!defined('ABSPATH')) {
    exit;
}

class NPK_Fertilizer_Selector_Cache {
    
    const PREFIX = 'npk_fertilizer_';
    const DEFAULT_EXPIRATION = 3600; // 1 час
    
    /**
     * Сохранить данные в кеш
     *
     * @param string $key Уникальный ключ
     * @param mixed $data Данные для сохранения
     * @param int $expiration Время жизни в секундах
     * @return bool
     */
    public function set($key, $data, $expiration = self::DEFAULT_EXPIRATION) {
        $transient_key = self::PREFIX . md5($key);
        return set_transient($transient_key, $data, $expiration);
    }
    
    /**
     * Получить данные из кеша
     *
     * @param string $key Уникальный ключ
     * @return mixed|false Данные или false если нет в кеше
     */
    public function get($key) {
        $transient_key = self::PREFIX . md5($key);
        return get_transient($transient_key);
    }
    
    /**
     * Инвалидировать конкретный ключ
     *
     * @param string $key Уникальный ключ
     * @return bool
     */
    public function invalidate($key) {
        $transient_key = self::PREFIX . md5($key);
        return delete_transient($transient_key);
    }
    
    /**
     * Инвалидировать все ключи модуля
     *
     * @return bool
     */
    public function invalidate_all() {
        global $wpdb;
        
        // Ищем все transients с нашим префиксом
        $options = $wpdb->get_col(
            $wpdb->prepare(
                "SELECT option_name FROM {$wpdb->options} 
                WHERE option_name LIKE %s 
                OR option_name LIKE %s",
                '_transient_' . self::PREFIX . '%',
                '_transient_timeout_' . self::PREFIX . '%'
            )
        );
        
        foreach ($options as $option) {
            delete_option($option);
        }
        
        return true;
    }
    
    /**
     * Получить ключ для конкретного запроса
     *
     * @param string $crop_id ID культуры
     * @param string $phase_id ID фазы
     * @return string
     */
    public function get_query_key($crop_id, $phase_id) {
        return "query_{$crop_id}_{$phase_id}";
