    /**
     * НОРМАЛИЗАЦИЯ NPK-ВЕКТОРА
     * 
     * Приводит вектор к единичной длине (нормализация).
     * Используется модулем подбора удобрений (Selection_Engine).
     * 
     * @param array $npk [N, P, K] значения
     * @return array Нормализованный вектор
     * 
     * @since 4.1 Доступ public (ранее private)
     */
    public static function normalize_npk($npk) {
        $sum = array_sum($npk);
        if ($sum <= 0) {
            return [0, 0, 0];
        }
        return [
            $npk[0] / $sum,
            $npk[1] / $sum,
            $npk[2] / $sum
        ];
    }
