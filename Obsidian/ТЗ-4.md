**РОЛЬ:** Транспортный слой. Принимает JSON-запросы от UI, валидирует их, вызывает `Calculation_Engine` и возвращает JSON-ответ.

**ЗАДАЧА:** Реализовать полноценный AJAX обработчик согласно ТЗ с учётом архитектурных ограничений.

**КОД (РЕАЛИЗАЦИЯ):**
```php
<?php
/**
 * AJAX Handler - обработка AJAX запросов
 * ТЗ-4: Полная реализация
 */

if (!defined('ABSPATH')) {
    exit;
}

final class NPK_Concentration_AJAX_Handler {
    
    /**
     * Обработчик AJAX запросов
     * Формат запроса: строго JSON согласно ТЗ п.7.3.2
     * Формат ответа: строго JSON согласно ТЗ п.7.3.3-7.3.4
     */
    public static function handle_request(): void {
        // 1. Устанавливаем заголовок Content-Type для JSON
        header('Content-Type: application/json; charset=utf-8');
        
        // 2. Проверяем nonce для безопасности
        if (!self::verify_nonce()) {
            self::send_error_response('Ошибка безопасности: недействительный nonce.', 403);
            return;
        }
        
        // 3. Получаем и парсим JSON данные
        $json_input = file_get_contents('php://input');
        $data = json_decode($json_input, true);
        
        if (json_last_error() !== JSON_ERROR_NONE) {
            self::send_error_response('Неверный формат JSON запроса.', 400);
            return;
        }
        
        // 4. Валидация входных данных
        $validated_data = self::validate_request_data($data);
        if (!$validated_data) {
            self::send_error_response('Неверные или отсутствующие параметры запроса.', 400);
            return;
        }
        
        // 5. Вызываем действие WordPress хука перед расчётом
        do_action('npk_concentration_before_calculate', $validated_data);
        
        try {
            // 6. Вызываем Calculation Engine для расчёта
            $result = NPK_Concentration_Calculation_Engine::calculate(
                $validated_data['fertilizer_id'],
                $validated_data['concentration'],
                $validated_data['water_volume'],
                $validated_data['ground_type']
            );
            
            // 7. Применяем фильтр WordPress для модификации результата
            $result = apply_filters('npk_concentration_calculated_result', $result, $validated_data);
            
            // 8. Вызываем действие WordPress хука после расчёта
            do_action('npk_concentration_after_calculate', $result, $validated_data);
            
            // 9. Отправляем успешный ответ
            self::send_success_response($result);
            
        } catch (Exception $e) {
            // 10. Обработка исключений Calculation Engine
            error_log('NPK Calculation Engine Error: ' . $e->getMessage());
            self::send_error_response('Внутренняя ошибка сервера при расчёте.', 500);
        }
    }
    
    /**
     * Проверка nonce для безопасности
     * @return bool Результат проверки
     */
    private static function verify_nonce(): bool {
        // Получаем nonce из заголовков или из данных запроса
        $nonce = '';
        
        // Проверяем заголовок X-WP-Nonce
        if (isset($_SERVER['HTTP_X_WP_NONCE'])) {
            $nonce = sanitize_text_field(wp_unslash($_SERVER['HTTP_X_WP_NONCE']));
        }
        // Или из JSON данных
        elseif (isset($_POST['nonce'])) {
            $nonce = sanitize_text_field(wp_unslash($_POST['nonce']));
        }
        
        return wp_verify_nonce($nonce, 'npk_concentration_calculate');
    }
    
    /**
     * Валидация входных данных согласно ТЗ п.7.3.2
     * @param array $data Данные запроса
     * @return array|null Валидированные данные или null при ошибке
     */
    private static function validate_request_data(array $data): ?array {
        // Проверяем обязательные поля согласно ТЗ п.7.3.2
        $required_fields = ['fertilizer_id', 'concentration', 'water_volume', 'ground_type'];
        
        foreach ($required_fields as $field) {
            if (!isset($data[$field])) {
                return null;
            }
        }
        
        // Валидация fertilizer_id (должен существовать в БД)
        $fertilizer_id = filter_var($data['fertilizer_id'], FILTER_VALIDATE_INT);
        if ($fertilizer_id === false || $fertilizer_id <= 0) {
            return null;
        }
        
        // Проверка существования удобрения в БД
        if (!self::fertilizer_exists($fertilizer_id)) {
            return null;
        }
        
        // Валидация concentration (0.05 - 1.5 %)
        $concentration = filter_var($data['concentration'], FILTER_VALIDATE_FLOAT);
        if ($concentration === false || $concentration < 0.05 || $concentration > 1.5) {
            return null;
        }
        
        // Валидация water_volume (0.5 - 200 л)
        $water_volume = filter_var($data['water_volume'], FILTER_VALIDATE_FLOAT);
        if ($water_volume === false || $water_volume < 0.5 || $water_volume > 200) {
            return null;
        }
        
        // Валидация ground_type (только допустимые значения)
        $ground_type = sanitize_text_field($data['ground_type']);
        $allowed_ground_types = ['dry', 'medium', 'wet'];
        if (!in_array($ground_type, $allowed_ground_types, true)) {
            return null;
        }
        
        return [
            'fertilizer_id' => $fertilizer_id,
            'concentration' => $concentration,
            'water_volume' => $water_volume,
            'ground_type' => $ground_type
        ];
    }
    
    /**
     * Проверка существования удобрения в БД
     * @param int $fertilizer_id ID удобрения
     * @return bool Существует ли удобрение
     */
    private static function fertilizer_exists(int $fertilizer_id): bool {
        // Проверяем существование класса БД
        if (!class_exists('NPK_Database')) {
            return false;
        }
        
        $fertilizers = NPK_Database::get_fertilizers();
        
        foreach ($fertilizers as $fertilizer) {
            if ($fertilizer['id'] === $fertilizer_id) {
                return true;
            }
        }
        
        return false;
    }
    
    /**
     * Отправка успешного JSON ответа согласно ТЗ п.7.3.3
     * @param array $result Результат расчёта из Calculation Engine
     */
    private static function send_success_response(array $result): void {
        $response = [
            'success' => true,
            'data' => $result
        ];
        
        wp_send_json($response, 200);
    }
    
    /**
     * Отправка ошибочного JSON ответа согласно ТЗ п.7.3.4
     * @param string $message Сообщение об ошибке
     * @param int $status_code HTTP статус код
     */
    private static function send_error_response(string $message, int $status_code = 400): void {
        // Устанавливаем HTTP статус код
        status_header($status_code);
        
        $response = [
            'success' => false,
            'data' => [
                'message' => $message
            ]
        ];
        
        wp_send_json($response, $status_code);
    }
}
```
