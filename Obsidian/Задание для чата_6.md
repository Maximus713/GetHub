spl_autoload_register( function( $class ) {

    $prefix = 'NPK_Selector_';
    $base_dir = __DIR__ . '/includes/';
    
    $len = strlen( $prefix );
    if ( strncmp( $prefix, $class, $len ) !== 0 ) {
        return;

    $relative_class = substr( $class, $len );
    
    // Конвертируем из CamelCase в snake-case для имени файла
    // NPK_Selector_Cache_Manager → class-cache-manager.php
    $file_name = 'class-' . str_replace( '_', '-', strtolower( $relative_class ) ) . '.php';
    $file = $base_dir . $file_name;
    
    if ( file_exists( $file ) ) {
        require $file;
    }

<?php
/**
 * Класс для управления кешем модуля подбора удобрений
 *
 * @package NPK_Calculator
 * @subpackage Fertilizer_Selector
 * @since 1.0.0
 */

// Защита от прямого доступа
if ( ! defined( 'ABSPATH' ) ) {
    exit;
}

/**
 * Класс кеш-менеджера
 * 
 * @uses WP Transients API для хранения данных
 */
class NPK_Selector_Cache_Manager {

    /**
     * Префикс для всех ключей кеша
     */
    const CACHE_PREFIX = 'npk_selector_';

    /**
     * Сохраняет данные в кеш
     *
     * @param string $key    Ключ кеша (без префикса)
     * @param mixed  $data   Данные для сохранения
     * @param int    $expire Время жизни кеша в секундах (по умолчанию 3600)
     * @return bool          Результат сохранения
     */
    public function set( $key, $data, $expire = 3600 ) {
        if ( empty( $key ) ) {
            return false;
        }

        $full_key = self::CACHE_PREFIX . $key;
        return set_transient( $full_key, $data, $expire );
    }

    /**
     * Получает данные из кеша
     *
     * @param string $key Ключ кеша (без префикса)
     * @return mixed|false Данные из кеша или false при отсутствии/истечении
     */
    public function get( $key ) {
        if ( empty( $key ) ) {
            return false;
        }

        $full_key = self::CACHE_PREFIX . $key;
        return get_transient( $full_key );
    }

    /**
     * Инвалидирует кеш
     *
     * @param int|null $crop_id ID культуры для удаления специфичного кеша
     * @return bool Результат операции
     */
    public function invalidate( $crop_id = null ) {
        global $wpdb;

        if ( null !== $crop_id ) {
            // Удаляем кеш для конкретной культуры и всех её фаз
            $pattern = self::CACHE_PREFIX . '%' . $wpdb->esc_like( md5( $crop_id . '_' ) ) . '%';
            
            $keys = $wpdb->get_col(
                $wpdb->prepare(
                    "SELECT option_name FROM {$wpdb->options} 
                    WHERE option_name LIKE %s 
                    AND option_name NOT LIKE %s",
                    $pattern,
                    '_transient_timeout_%'
                )
            );

            $result = true;
            foreach ( $keys as $key ) {
                // Удаляем префикс 'npk_selector_' из начала ключа
                $clean_key = preg_replace( '/^' . preg_quote( self::CACHE_PREFIX, '/' ) . '/', '', $key );
                if ( ! delete_transient( $clean_key ) ) {
                    $result = false;
                }
            }
            
            return $result;
        } else {
            // Удаляем ВЕСЬ кеш модуля
            $pattern = self::CACHE_PREFIX . '%';
            
            $keys = $wpdb->get_col(
                $wpdb->prepare(
                    "SELECT option_name FROM {$wpdb->options} 
                    WHERE option_name LIKE %s 
                    AND option_name NOT LIKE %s",
                    $pattern,
                    '_transient_timeout_%'
                )
            );

            $result = true;
            foreach ( $keys as $key ) {
                // Удаляем префикс 'npk_selector_' из начала ключа
                $clean_key = preg_replace( '/^' . preg_quote( self::CACHE_PREFIX, '/' ) . '/', '', $key );
                if ( ! delete_transient( $clean_key ) ) {
                    $result = false;
                }
            }
            
            return $result;
        }
    }

    /**
     * Генерирует ключ кеша для культуры и фазы
     *
     * @param int|string $crop_id  ID культуры
     * @param int|string $phase_id ID фазы (опционально)
     * @return string              Сгенерированный ключ
     */
    public function generate_key( $crop_id, $phase_id = '' ) {
        if ( empty( $phase_id ) ) {
            return md5( $crop_id );
        }
        return md5( $crop_id . '_' . $phase_id );
    }

    /**
     * Пример использования с БД (для документации)
     * Сам метод не используется, но показывает как правильно работать с таблицей
     * 
     * @return void
     */
    private function _example_db_usage() {
        global $wpdb;
        
        // Правильное обращение к существующей таблице
        $table = $wpdb->prefix . 'crop_phase_fertilizer';
        
        // Всегда используем prepare для запросов с параметрами
        $crop_id = 5;
        $results = $wpdb->get_results(
            $wpdb->prepare(
                "SELECT * FROM {$table} WHERE crop_id = %d",
                $crop_id
            )
        );
        
        // Используем кеш для результатов
        $cache_key = $this->generate_key( $crop_id, 'fertilizers' );
        if ( false === $this->get( $cache_key ) ) {
            // Данных нет в кеше - получаем из БД и сохраняем
            $this->set( $cache_key, $results, 3600 );
        }
    }
}

1. **Главный файл модуля (`fertilizer-selector-module.php`):**
   - В самом начале файла выполнить проверку: `if ( ! class_exists( 'NPK_Calculators_Core' ) )`
   - Если класс ядра отсутствует:
     - НЕ инициализировать модуль (прервать выполнение с `return;`)

       `<div class="notice notice-warning is-dismissible"><p>Модуль подбора удобрений требует активации основного плагина NPK Calculators.</p></div>`
   - Если класс ядра существует:
     - Запустить инициализацию модуля (подключение файлов, автозагрузка, хуки)
     - Использовать хук `plugins_loaded` для инициализации, чтобы гарантировать загрузку всех зависимостей

2. **Автозагрузка классов:**
   - Зарегистрировать автозагрузчик через `spl_autoload_register`
   - Префикс классов: `NPK_Selector_`
   - Маппинг: `NPK_Selector_Class_Name` → `includes/class-class-name.php`
   - Пример: `NPK_Selector_Cache_Manager` → `includes/class-cache-manager.php`

3. **Класс `NPK_Selector_Cache_Manager`:**
   - Реализовать методы:
     - `set( $key, $data, $expire = 3600 )` — сохраняет данные в транзиты WordPress
     - `get( $key )` — возвращает данные из кеша или false при отсутствии/истечении
     - `invalidate( $crop_id = null )` — удаляет кеш по конкретной культуре или весь кеш модуля
   - Формирование ключей:
     - Для всех ключей использовать префикс: `'npk_selector_'`
     - Для ключей с параметрами: `'npk_selector_' . md5( $crop_id . '_' . $phase_id )`
   - Внутри использовать нативные функции WordPress: `set_transient()`, `get_transient()`, `delete_transient()`

4. **Интеграция с БД:**
   - НЕ создавать новых таблиц
   - Использовать существующую таблицу `crop_phase_fertilizer` из основной версии 4.0
   - Получать объект БД через глобальную переменную: `global $wpdb; $table = $wpdb->prefix . 'crop_phase_fertilizer';`

- ❌ **Неправильные имена файлов (Ошибка 1)** — пример: файл назван `class-cachemanager.php` вместо `class-cache-manager.php` → строго соблюдаем WordPress-стандарт: `class-{имя-класса}.php` с дефисами, все в нижнем регистре

- ❌ **Нарушение порядка загрузки (Ошибка 3)** — пример: попытка использовать классы до их подключения → все инициализации только после `plugins_loaded`, проверка существования ядра до любого кода модуля
